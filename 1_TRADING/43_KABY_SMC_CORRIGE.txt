//@version=6
// KABY SMC — Order Blocks MTF (version corrigée).
// Corrections : point final en trop supprimé ; zones dessinées à partir de la VRAIE bougie OB (temps, pas index) ;
// zone supprimée dès qu'une bougie clôture au travers ; un sommet / creux cassé ne sert qu'une fois ;
// FVG vérifié JUSTE APRÈS l'OB ; plusieurs zones gardées par sens ; 1 seule alerte au 1er retest de chaque zone ;
// le nom de l'unité suit le réglage (H1, H4…). À utiliser sur une unité PLUS PETITE que celle des OB (ex. M5 avec OB H1).
indicator("KABY SMC - Order Blocks MTF (corrigé)", shorttitle = "KABY SMC", overlay = true,
     max_boxes_count = 60, max_labels_count = 100, max_lines_count = 60)

//==================================================
// 1. PARAMETRES
//==================================================
obTf = input.timeframe("60", "Unité de temps des Order Blocks")
swingLen = input.int(5, "Force des swings", minval = 2, maxval = 20)
obSearch = input.int(8, "Bougies source à rechercher", minval = 2, maxval = 20)
atrLen = input.int(14, "Période ATR", minval = 2)
impulseMult = input.float(1.0, "Force minimale de l'impulsion (× ATR)", minval = 0.2, step = 0.1)
sweepLookback = input.int(10, "Validité de la prise de liquidité (bougies)", minval = 1)
minScore = input.int(3, "Score minimum", minval = 2, maxval = 4)
requireSweep = input.bool(false, "Exiger une prise de liquidité")
requireFVG = input.bool(false, "Exiger un FVG juste après l'OB")
useBody = input.bool(false, "Utiliser le corps de la bougie source")
keepN = input.int(3, "Zones gardées par sens", minval = 1, maxval = 10)
showLabels = input.bool(true, "Afficher les labels")
showMid = input.bool(true, "Afficher le milieu des zones")
showTouches = input.bool(true, "Afficher le 1er retest")
showDashboard = input.bool(true, "Afficher le tableau")

tfName = obTf == "60" ? "H1" : obTf == "240" ? "H4" : obTf == "D" ? "Jour" : obTf == "W" ? "Semaine" : obTf + " min"

//==================================================
// 2. DETECTION SUR L'UNITE DE TEMPS DES OB
//==================================================
f_detect() =>
    atr = ta.atr(atrLen)
    ph = ta.pivothigh(high, swingLen, swingLen)
    pl = ta.pivotlow(low, swingLen, swingLen)
    var float lastPH = na
    var float lastPL = na
    var bool phUsed = true
    var bool plUsed = true
    if not na(ph)
        lastPH := ph
        phUsed := false
    if not na(pl)
        lastPL := pl
        plUsed := false

    sweepLow = not na(lastPL) and low < lastPL and close > lastPL
    sweepHigh = not na(lastPH) and high > lastPH and close < lastPH
    sinceLow = ta.barssince(sweepLow)
    sinceHigh = ta.barssince(sweepHigh)
    recentSweepLow = not na(sinceLow) and sinceLow <= sweepLookback
    recentSweepHigh = not na(sinceHigh) and sinceHigh <= sweepLookback

    bullImpulse = close > open and not na(atr) and close - open >= atr * impulseMult
    bearImpulse = close < open and not na(atr) and open - close >= atr * impulseMult
    // un sommet / creux cassé ne sert qu'une fois
    bosUp = not phUsed and not na(lastPH) and close > lastPH
    bosDn = not plUsed and not na(lastPL) and close < lastPL

    float bTop = na
    float bBot = na
    int   bT = na
    int   bScore = 0
    bool  newBull = false
    float sTop = na
    float sBot = na
    int   sT = na
    int   sScore = 0
    bool  newBear = false

    if bosUp
        phUsed := true
        int j = na
        for i = 1 to obSearch
            if na(j) and close[i] < open[i]
                j := i
        if not na(j) and bullImpulse
            // FVG juste après l'OB (3 bougies qui partent de l'OB, ou les 3 suivantes)
            fvgOk = (j >= 2 and low[j - 2] > high[j]) or (j >= 3 and low[j - 3] > high[j - 1])
            score = 2 + (recentSweepLow ? 1 : 0) + (fvgOk ? 1 : 0)
            if score >= minScore and (not requireSweep or recentSweepLow) and (not requireFVG or fvgOk)
                bTop := useBody ? math.max(open[j], close[j]) : high[j]
                bBot := useBody ? math.min(open[j], close[j]) : low[j]
                bT := time[j]
                bScore := score
                newBull := true

    if bosDn
        plUsed := true
        int j = na
        for i = 1 to obSearch
            if na(j) and close[i] > open[i]
                j := i
        if not na(j) and bearImpulse
            fvgOk = (j >= 2 and high[j - 2] < low[j]) or (j >= 3 and high[j - 3] < low[j - 1])
            score = 2 + (recentSweepHigh ? 1 : 0) + (fvgOk ? 1 : 0)
            if score >= minScore and (not requireSweep or recentSweepHigh) and (not requireFVG or fvgOk)
                sTop := useBody ? math.max(open[j], close[j]) : high[j]
                sBot := useBody ? math.min(open[j], close[j]) : low[j]
                sT := time[j]
                sScore := score
                newBear := true

    // valeurs de la bougie précédente (clôturée) : rien ne se redessine après coup
    [bTop[1], bBot[1], bT[1], newBull[1], bScore[1], sTop[1], sBot[1], sT[1], newBear[1], sScore[1]]

[bTop, bBot, bT, bEv, bSc, sTop, sBot, sT, sEv, sSc] = request.security(syminfo.tickerid, obTf, f_detect(), gaps = barmerge.gaps_off, lookahead = barmerge.lookahead_on)

newHtfBar = timeframe.change(obTf)
newBull = newHtfBar and bEv and not na(bTop)
newBear = newHtfBar and sEv and not na(sTop)

//==================================================
// 3. ZONES (dessinées en temps : elles suivent le graphique)
//==================================================
type Z
    int   dir
    float top
    float bot
    box   bx
    line  md
    int   born
    bool  touched = false

var array<Z> zs = array.new<Z>()

if newBull
    bx = box.new(bT, bTop, time, bBot, xloc = xloc.bar_time, extend = extend.right, border_color = color.lime, bgcolor = color.new(color.lime, 86))
    line md = na
    if showMid
        md := line.new(bT, (bTop + bBot) / 2, time, (bTop + bBot) / 2, xloc = xloc.bar_time, extend = extend.right, color = color.lime, style = line.style_dotted)
    zs.push(Z.new(1, bTop, bBot, bx, md, bar_index))
    if showLabels
        label.new(time, bBot, "OB " + tfName + " ACHAT | Score " + str.tostring(bSc) + "/4", xloc = xloc.bar_time,
             style = label.style_label_up, color = color.green, textcolor = color.white, size = size.small)

if newBear
    bx = box.new(sT, sTop, time, sBot, xloc = xloc.bar_time, extend = extend.right, border_color = color.red, bgcolor = color.new(color.red, 86))
    line md = na
    if showMid
        md := line.new(sT, (sTop + sBot) / 2, time, (sTop + sBot) / 2, xloc = xloc.bar_time, extend = extend.right, color = color.red, style = line.style_dotted)
    zs.push(Z.new(-1, sTop, sBot, bx, md, bar_index))
    if showLabels
        label.new(time, sTop, "OB " + tfName + " VENTE | Score " + str.tostring(sSc) + "/4", xloc = xloc.bar_time,
             style = label.style_label_down, color = color.red, textcolor = color.white, size = size.small)

// garder les keepN zones les plus récentes par sens
for d in array.from(1, -1)
    int cnt = 0
    for z in zs
        cnt += z.dir == d ? 1 : 0
    if cnt > keepN
        for i = 0 to zs.size() - 1
            if zs.get(i).dir == d
                zr = zs.remove(i)
                box.delete(zr.bx)
                if not na(zr.md)
                    line.delete(zr.md)
                break

//==================================================
// 4. RETESTS ET INVALIDATION
//==================================================
bool bullTouch = false
bool bearTouch = false
if zs.size() > 0
    for i = zs.size() - 1 to 0
        z = zs.get(i)
        dead = z.dir == 1 ? close < z.bot : close > z.top
        if dead
            // zone cassée en clôture : supprimée (plus de faux retests)
            zr = zs.remove(i)
            box.delete(zr.bx)
            if not na(zr.md)
                line.delete(zr.md)
        else if not z.touched and bar_index > z.born and (z.dir == 1 ? low <= z.top : high >= z.bot)
            z.touched := true
            box.set_bgcolor(z.bx, color.new(z.dir == 1 ? color.lime : color.red, 93))
            if z.dir == 1
                bullTouch := true
            else
                bearTouch := true

plotshape(showTouches and bullTouch, title = "1er retest OB haussier", style = shape.circle, location = location.belowbar, color = color.lime, size = size.tiny)
plotshape(showTouches and bearTouch, title = "1er retest OB baissier", style = shape.circle, location = location.abovebar, color = color.red, size = size.tiny)

//==================================================
// 5. ALERTES (une seule par événement)
//==================================================
alertcondition(newBull, title = "Nouvel OB haussier", message = "Nouvel Order Block haussier sur {{ticker}}.")
alertcondition(newBear, title = "Nouvel OB baissier", message = "Nouvel Order Block baissier sur {{ticker}}.")
alertcondition(bullTouch, title = "1er retest OB haussier", message = "Le prix touche pour la 1re fois un Order Block haussier sur {{ticker}}.")
alertcondition(bearTouch, title = "1er retest OB baissier", message = "Le prix touche pour la 1re fois un Order Block baissier sur {{ticker}}.")

//==================================================
// 6. TABLEAU
//==================================================
var table dash = table.new(position.top_right, 2, 4, border_width = 1)
if barstate.islast and showDashboard
    float nbT = na
    float nbB = na
    float nsT = na
    float nsB = na
    for z in zs
        if z.dir == 1 and z.top < close and (na(nbT) or z.top > nbT)
            nbT := z.top
            nbB := z.bot
        if z.dir == -1 and z.bot > close and (na(nsB) or z.bot < nsB)
            nsT := z.top
            nsB := z.bot
    table.cell(dash, 0, 0, "KABY SMC", bgcolor = color.black, text_color = color.white)
    table.cell(dash, 1, 0, tfName + " / MTF", bgcolor = color.black, text_color = color.white)
    table.cell(dash, 0, 1, "OB haussier le + proche")
    table.cell(dash, 1, 1, na(nbT) ? "Aucun" : str.tostring(nbB, format.mintick) + " - " + str.tostring(nbT, format.mintick), text_color = color.green)
    table.cell(dash, 0, 2, "OB baissier le + proche")
    table.cell(dash, 1, 2, na(nsT) ? "Aucun" : str.tostring(nsB, format.mintick) + " - " + str.tostring(nsT, format.mintick), text_color = color.red)
    table.cell(dash, 0, 3, "Score minimum")
    table.cell(dash, 1, 3, str.tostring(minScore) + "/4")
