//@version=6
// PRICE ACTION 4 — FIGURES CHARTISTES (patterns)
// Calcul sur la référence FIXE M15 (mêmes figures de M1 à H1 ; H4 / jour : unité affichée), à partir des sommets / creux
// confirmés (N bougies de chaque côté), reliés en zigzag.
// Figures détectées (la plus récente de chaque famille) :
//  RETOURNEMENT
//  • Double sommet / double creux : 2 sommets (creux) au même niveau (± 0,5 ATR), séparés d'un creux (sommet) d'au moins 1,5 ATR.
//    Ligne de cou = ce creux (sommet). CONFIRMÉ = clôture de l'autre côté de la ligne de cou. Objectif = hauteur reportée.
//  • Épaule-tête-épaule (et inversée) : tête plus haute (basse) que les 2 épaules (≥ 0,5 ATR), épaules au même niveau (± 1 ATR).
//    Ligne de cou = les 2 creux (sommets) entre épaules et tête. Confirmation et objectif idem.
//  CONTINUATION / COMPRESSION (2 derniers sommets + 2 derniers creux)
//  • Triangle symétrique, ascendant, descendant ; biseau montant (souvent baissier) ; biseau descendant (souvent haussier).
//    Sortie = clôture au-dessus de la ligne haute ou sous la ligne basse.
// En formation = pointillés ; confirmé / sorti = trait plein + objectif. Figure invalidée = effacée. Ne passe aucun ordre.
indicator("Price action 4 — Figures chartistes", shorttitle = "PA FIGURES", overlay = true, max_lines_count = 40, max_labels_count = 20, max_bars_back = 500)

srcTf  = input.string("15", "Unité de référence (mêmes figures jusqu'au H1)", options = ["15", "60", "240", "Unité affichée"])
pivLen = input.int(5, "Sommet / creux confirmé par N bougies de chaque côté", minval = 2, maxval = 20)
dtTol  = input.float(0.5, "Double sommet / creux : écart max entre les 2 (× ATR)", minval = 0.1, step = 0.1)
dtDep  = input.float(1.5, "Double sommet / creux : profondeur min (× ATR)", minval = 0.5, step = 0.25)
flatK  = input.float(0.3, "Triangle : ligne « plate » si elle bouge de moins de (× ATR)", minval = 0.05, step = 0.05)
extH   = input.float(6.0, "Prolonger à droite (heures)", minval = 1, step = 1)
showDT = input.bool(true, "Double sommet / double creux")
showHS = input.bool(true, "Épaule-tête-épaule")
showTR = input.bool(true, "Triangles et biseaux")

refTf = srcTf == "Unité affichée" or timeframe.in_seconds() > 3600 ? timeframe.period : srcTf

// Niveau d'une droite (b1, p1) → (b2, p2) à la bougie b
lineAt(int b1, float p1, int b2, float p2, int b) => p1 + (p2 - p1) / math.max(1, b2 - b1) * (b - b1)

// Statut après la bougie bS : 1 = clôture au-dessus de la droite haute, 2 = clôture sous la droite basse (1re qui arrive)
// Sortie : [statut, temps de la cassure]
breakAfter(int bS, int hb1, float hp1, int hb2, float hp2, int lb1, float lp1, int lb2, float lp2) =>
    int st = 0
    int bt = 0
    n = math.min(bar_index - bS, 450)
    if n > 0
        for k = n - 1 to 0
            b = bar_index - k
            if close[k] > lineAt(hb1, hp1, hb2, hp2, b)
                st := 1
                bt := time[k]
                break
            if close[k] < lineAt(lb1, lp1, lb2, lp2, b)
                st := 2
                bt := time[k]
                break
    [st, bt]

// Sortie : suite de figures de 14 valeurs : code, statut, t1, p1, t2, p2, t3, p3, t4, p4, t5, p5, objectif, temps de cassure
// codes : 1 double sommet, 2 double creux, 3 ETE, 4 ETE inversée, 5 triangle sym., 6 triangle asc., 7 triangle desc.,
//         8 biseau montant, 9 biseau descendant. Statut : 0 en formation, 1 sortie haute, 2 sortie basse.
patData() =>
    hp = ta.pivothigh(high, pivLen, pivLen)
    lp = ta.pivotlow(low, pivLen, pivLen)
    a = ta.atr(14)
    var array<float> sP = array.new<float>()
    var array<int>   sB = array.new<int>()
    var array<int>   sT = array.new<int>()
    var array<int>   sY = array.new<int>()
    // zigzag : sommets et creux en alternance (2 du même type de suite → on garde le plus extrême)
    if not na(hp)
        if sY.size() > 0 and sY.last() == 1
            if hp > sP.last()
                sP.set(sP.size() - 1, hp)
                sB.set(sB.size() - 1, bar_index - pivLen)
                sT.set(sT.size() - 1, time[pivLen])
        else
            sP.push(hp)
            sB.push(bar_index - pivLen)
            sT.push(time[pivLen])
            sY.push(1)
    if not na(lp)
        if sY.size() > 0 and sY.last() == -1
            if lp < sP.last()
                sP.set(sP.size() - 1, lp)
                sB.set(sB.size() - 1, bar_index - pivLen)
                sT.set(sT.size() - 1, time[pivLen])
        else
            sP.push(lp)
            sB.push(bar_index - pivLen)
            sT.push(time[pivLen])
            sY.push(-1)
    while sP.size() > 12
        sP.shift()
        sB.shift()
        sT.shift()
        sY.shift()
    out = array.new<float>()
    n = sP.size()
    if barstate.islast and n >= 3
        // ---- Double sommet / double creux : 3 derniers points (ou 3 points avant le dernier) ----
        if showDT
            bool found = false
            for e = n - 1 to math.max(2, n - 2)
                if not found
                    p1 = sP.get(e - 2)
                    p2 = sP.get(e - 1)
                    p3 = sP.get(e)
                    y = sY.get(e)
                    if math.abs(p1 - p3) <= dtTol * a and (y == 1 ? math.min(p1, p3) - p2 : p2 - math.max(p1, p3)) >= dtDep * a
                        ext = y == 1 ? math.max(p1, p3) : math.min(p1, p3)
                        // ligne de cou horizontale = p2 ; invalidé si clôture au-delà du sommet (creux)
                        [st, bt] = breakAfter(sB.get(e), sB.get(e), y == 1 ? ext : p2, sB.get(e) + 1, y == 1 ? ext : p2, sB.get(e), y == 1 ? p2 : ext, sB.get(e) + 1, y == 1 ? p2 : ext)
                        bool inval = y == 1 ? st == 1 : st == 2
                        if not inval
                            found := true
                            h = math.abs((p1 + p3) / 2 - p2)
                            out.push(y == 1 ? 1 : 2)
                            out.push(st)
                            out.push(sT.get(e - 2))
                            out.push(p1)
                            out.push(sT.get(e - 1))
                            out.push(p2)
                            out.push(sT.get(e))
                            out.push(p3)
                            for q = 1 to 4
                                out.push(0)
                            out.push(y == 1 ? p2 - h : p2 + h)
                            out.push(bt)
        // ---- Épaule-tête-épaule : 5 points ----
        if showHS and n >= 5
            bool found = false
            for e = n - 1 to math.max(4, n - 2)
                if not found
                    y = sY.get(e)
                    s1 = sP.get(e - 4)
                    n1 = sP.get(e - 3)
                    hd = sP.get(e - 2)
                    n2 = sP.get(e - 1)
                    s2 = sP.get(e)
                    bool shape = y == 1 ? hd - math.max(s1, s2) >= 0.5 * a : math.min(s1, s2) - hd >= 0.5 * a
                    if shape and math.abs(s1 - s2) <= 1.0 * a
                        nb1 = sB.get(e - 3)
                        nb2 = sB.get(e - 1)
                        // ligne de cou (n1 → n2) ; invalidé si clôture au-delà de la tête
                        be = sB.get(e)
                        [st, bt] = breakAfter(be, y == 1 ? be : nb1, y == 1 ? hd : n1, y == 1 ? be + 1 : nb2, y == 1 ? hd : n2,
                             y == 1 ? nb1 : be, y == 1 ? n1 : hd, y == 1 ? nb2 : be + 1, y == 1 ? n2 : hd)
                        bool inval = y == 1 ? st == 1 : st == 2
                        if not inval
                            found := true
                            neckH = lineAt(nb1, n1, nb2, n2, sB.get(e - 2))
                            h = math.abs(hd - neckH)
                            neckNow = lineAt(nb1, n1, nb2, n2, bar_index)
                            out.push(y == 1 ? 3 : 4)
                            out.push(st)
                            out.push(sT.get(e - 4))
                            out.push(s1)
                            out.push(sT.get(e - 3))
                            out.push(n1)
                            out.push(sT.get(e - 2))
                            out.push(hd)
                            out.push(sT.get(e - 1))
                            out.push(n2)
                            out.push(sT.get(e))
                            out.push(s2)
                            out.push(y == 1 ? neckNow - h : neckNow + h)
                            out.push(bt)
        // ---- Triangles / biseaux : 2 derniers sommets et 2 derniers creux ----
        if showTR and n >= 4
            int h2 = sY.last() == 1 ? n - 1 : n - 2
            int l2 = sY.last() == -1 ? n - 1 : n - 2
            int h1 = h2 - 2
            int l1 = l2 - 2
            if h1 >= 0 and l1 >= 0
                hb1 = sB.get(h1)
                hb2 = sB.get(h2)
                lb1 = sB.get(l1)
                lb2 = sB.get(l2)
                hp1 = sP.get(h1)
                hp2 = sP.get(h2)
                lp1 = sP.get(l1)
                lp2 = sP.get(l2)
                dH = hp2 - hp1
                dL = lp2 - lp1
                b0 = math.min(hb1, lb1)
                bE = math.max(hb2, lb2)
                widthStart = lineAt(hb1, hp1, hb2, hp2, b0) - lineAt(lb1, lp1, lb2, lp2, b0)
                widthEnd = lineAt(hb1, hp1, hb2, hp2, bE) - lineAt(lb1, lp1, lb2, lp2, bE)
                flatH = math.abs(dH) < flatK * a
                flatL = math.abs(dL) < flatK * a
                converge = widthEnd < widthStart * 0.85 and widthEnd > 0
                int code = 0
                if converge
                    code := flatH and dL > 0 ? 6 : flatL and dH < 0 ? 7 : dH < 0 and dL > 0 ? 5 : dH > 0 and dL > 0 ? 8 : dH < 0 and dL < 0 ? 9 : 0
                if code > 0
                    [st, bt] = breakAfter(bE, hb1, hp1, hb2, hp2, lb1, lp1, lb2, lp2)
                    out.push(code)
                    out.push(st)
                    out.push(sT.get(h1))
                    out.push(hp1)
                    out.push(sT.get(h2))
                    out.push(hp2)
                    out.push(sT.get(l1))
                    out.push(lp1)
                    out.push(sT.get(l2))
                    out.push(lp2)
                    out.push(0)
                    out.push(0)
                    // objectif = largeur à l'ouverture, reportée depuis la sortie
                    out.push(st == 1 ? lineAt(hb1, hp1, hb2, hp2, bar_index) + widthStart : st == 2 ? lineAt(lb1, lp1, lb2, lp2, bar_index) - widthStart : 0)
                    out.push(bt)
    out

r = request.security(syminfo.tickerid, refTf, patData())

var array<line>  fLn = array.new<line>()
var array<label> fLb = array.new<label>()
cBear = #F23645
cBull = #089981
cTri  = #FF9800
fmt(float v) => str.tostring(v, format.mintick)

names = array.from("", "Double sommet", "Double creux", "Épaule-tête-épaule", "ETE inversée", "Triangle symétrique", "Triangle ascendant",
     "Triangle descendant", "Biseau montant", "Biseau descendant")

seg(int xA, float pa, int xB, float pb, color c, string sty, int w, bool ext) =>
    fLn.push(line.new(xA, pa, xB, pb, xloc = xloc.bar_time, color = c, style = sty, width = w, extend = ext ? extend.right : extend.none))
    w

drawFig(int o) =>
    code = int(r.get(o))
    st = int(r.get(o + 1))
    t1 = int(r.get(o + 2))
    p1 = r.get(o + 3)
    t2 = int(r.get(o + 4))
    p2 = r.get(o + 5)
    t3 = int(r.get(o + 6))
    p3 = r.get(o + 7)
    t4 = int(r.get(o + 8))
    p4 = r.get(o + 9)
    t5 = int(r.get(o + 10))
    p5 = r.get(o + 11)
    tgt = r.get(o + 12)
    tb = int(r.get(o + 13))
    tEnd = time + math.round(extH * 3600000)
    bool done = st != 0
    string sty = done ? line.style_solid : line.style_dotted
    if code <= 2
        c = code == 1 ? cBear : cBull
        seg(t1, p1, t2, p2, c, sty, 2, false)
        seg(t2, p2, t3, p3, c, sty, 2, false)
        seg(t2, p2, done ? tb : tEnd, p2, color.new(c, 20), line.style_dashed, 1, false)
        txt = names.get(code) + (done ? " CONFIRMÉ — objectif " + fmt(tgt) : " en formation — confirmé si clôture " + (code == 1 ? "sous " : "au-dessus de ") + fmt(p2))
        fLb.push(label.new(t3, p3, txt, xloc = xloc.bar_time, style = code == 1 ? label.style_label_down : label.style_label_up, color = color.new(c, 25), textcolor = color.white, size = size.small))
        if done
            seg(tb, tgt, tEnd, tgt, c, line.style_dashed, 1, false)
    else if code <= 4
        c = code == 3 ? cBear : cBull
        seg(t1, p1, t2, p2, c, sty, 2, false)
        seg(t2, p2, t3, p3, c, sty, 2, false)
        seg(t3, p3, t4, p4, c, sty, 2, false)
        seg(t4, p4, t5, p5, c, sty, 2, false)
        // ligne de cou prolongée
        fLn.push(line.new(t2, p2, t4, p4, xloc = xloc.bar_time, color = color.new(c, 20), style = line.style_dashed, extend = extend.right))
        neckTxt = code == 3 ? "sous la ligne de cou" : "au-dessus de la ligne de cou"
        txt = names.get(code) + (done ? " CONFIRMÉ — objectif " + fmt(tgt) : " en formation — confirmé si clôture " + neckTxt)
        fLb.push(label.new(t3, p3, txt, xloc = xloc.bar_time, style = code == 3 ? label.style_label_down : label.style_label_up, color = color.new(c, 25), textcolor = color.white, size = size.small))
        if done
            seg(tb, tgt, tEnd, tgt, c, line.style_dashed, 1, false)
    else
        c = code == 8 ? cBear : code == 9 ? cBull : cTri
        seg(t1, p1, t2, p2, c, sty, 2, not done)
        seg(t3, p3, t4, p4, c, sty, 2, not done)
        outTxt = st == 1 ? " — SORTIE PAR LE HAUT, objectif " + fmt(tgt) : st == 2 ? " — SORTIE PAR LE BAS, objectif " + fmt(tgt) : " — en formation : attendre une clôture hors du triangle"
        fLb.push(label.new(t4, p4, names.get(code) + outTxt, xloc = xloc.bar_time, style = label.style_label_up, color = color.new(c, 25), textcolor = color.white, size = size.small))
        if done
            seg(tb, tgt, tEnd, tgt, st == 1 ? cBull : cBear, line.style_dashed, 1, false)
    code

if barstate.islast
    for l in fLn
        line.delete(l)
    for l in fLb
        label.delete(l)
    fLn.clear()
    fLb.clear()
    if not na(r) and r.size() >= 14
        for f = 0 to int(r.size() / 14) - 1
            drawFig(f * 14)
