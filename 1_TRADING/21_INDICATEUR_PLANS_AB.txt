// ---------- Daily : état 20/10 + veille ----------
dayData() =>
    var int st = 0
    hiL = ta.highest(high, dLong)[1]
    loL = ta.lowest(low, dLong)[1]
    hiS = ta.highest(high, dShort)[1]
    loS = ta.lowest(low, dShort)[1]
    if close > hiL
        st := 1
    else if close < loL
        st := -1
    else if st == 1 and close < loS
        st := 0
    else if st == -1 and close > hiS
        st := 0
    [st[1], high[1], low[1], ta.atr(14)[1]]

h4Data() =>
    ph = ta.pivothigh(high, htfPiv, htfPiv)
    pl = ta.pivotlow(low, htfPiv, htfPiv)
    hA = ta.valuewhen(not na(ph), ph, 0)
    hB = ta.valuewhen(not na(ph), ph, 1)
    lA = ta.valuewhen(not na(pl), pl, 0)
    lB = ta.valuewhen(not na(pl), pl, 1)
    st = hA > hB and lA > lB ? 1 : hA < hB and lA < lB ? -1 : 0
    var array<float> uTop = array.new<float>()
    var array<float> uBot = array.new<float>()
    var array<float> dTop = array.new<float>()
    var array<float> dBot = array.new<float>()
    var array<int>   uT   = array.new<int>()
    var array<int>   dT   = array.new<int>()
    if array.size(uTop) > 0
        for i = array.size(uTop) - 1 to 0
            if low <= array.get(uBot, i)
                array.remove(uTop, i)
                array.remove(uBot, i)
                array.remove(uT, i)
    if array.size(dTop) > 0
        for i = array.size(dTop) - 1 to 0
            if high >= array.get(dTop, i)
                array.remove(dTop, i)
                array.remove(dBot, i)
                array.remove(dT, i)
    if low > high[2]
        array.push(uTop, low)
        array.push(uBot, high[2])
        array.push(uT, time[2])
    if high < low[2]
        array.push(dTop, low[2])
        array.push(dBot, high)
        array.push(dT, time[2])
    if array.size(uTop) > 60
        array.shift(uTop)
        array.shift(uBot)
        array.shift(uT)
    if array.size(dTop) > 60
        array.shift(dTop)
        array.shift(dBot)
        array.shift(dT)
    // FVG haussière non comblée sous le prix (aimant des ventes) : son haut ; FVG baissière au-dessus : son bas
    float below  = na
    float belowB = na
    int   belowT = na
    float above  = na
    float aboveT = na
    int   aboveTm = na
    if array.size(uTop) > 0
        for i = 0 to array.size(uTop) - 1
            t = array.get(uTop, i)
            if t < close and (na(below) or t > below)
                below  := t
                belowB := array.get(uBot, i)
                belowT := array.get(uT, i)
    if array.size(dBot) > 0
        for i = 0 to array.size(dBot) - 1
            b = array.get(dBot, i)
            if b > close and (na(above) or b < above)
                above   := b
                aboveT  := array.get(dTop, i)
                aboveTm := array.get(dT, i)
    [st[1], lA[1], hA[1], below[1], above[1], belowB[1], belowT[1], aboveT[1], aboveTm[1]]

        o := fvgAbove

// Un niveau du plan : trait + étiquette posée juste au-dessus du trait, À L'INTÉRIEUR du dessin (côté gauche)
planLine(int t0, int t1, float p, string txt, color c, int w) =>
    array.push(plL, line.new(t0, p, t1, p, xloc = xloc.bar_time, color = c, width = w))
    array.push(plT, label.new(t0 + math.round((t1 - t0) * 0.03), p, txt, xloc = xloc.bar_time, style = label.style_label_lower_left,
         color = color.new(c, 100), textcolor = c == color.white ? color.white : c, size = size.small))
// Un plan complet, dessiné de la même façon pour A et B :
// flèche prix → entrée, boîte risque (rouge) et gain (vert), traits SL / Entrée / TP1-TP2-TP3 (1, 2, 3 R, avant la cible) / Cible, flèche entrée → cible
drawPlan(int t0, int t1, float e, float sl, float tgt, string name, color c) =>
    dist = math.abs(e - sl)
    up   = tgt > e
    lots = lotsFor(dist)
    riskReal = lots * dist * 100 / eurUsd
    array.push(plL, line.new(time, close, t0, e, xloc = xloc.bar_time, color = color.new(c, 40), width = 2, style = line.style_arrow_right))
    array.push(plB, box.new(t0, math.max(e, sl), t1, math.min(e, sl), xloc = xloc.bar_time, border_color = color.new(cRed, 60), bgcolor = color.new(cRed, 82)))
    array.push(plB, box.new(t0, math.max(e, tgt), t1, math.min(e, tgt), xloc = xloc.bar_time, border_color = color.new(cGreen, 60), bgcolor = color.new(cGreen, 86)))
    array.push(plT, label.new(t0, math.min(sl, tgt, e), name + " | " + str.tostring(lots, "#.##") + " lot | risque " + str.tostring(riskReal, "#") + " €",
         xloc = xloc.bar_time, style = label.style_label_upper_left, color = c, textcolor = color.white, size = size.small))
    planLine(t0, t1, sl, "SL " + str.tostring(sl, format.mintick), cRed, 2)
    planLine(t0, t1, e, "Entrée " + str.tostring(e, format.mintick), color.white, 2)
    n = 0
    for k = 1 to 3
        tp = e + (up ? 1 : -1) * k * dist
        if up ? tp < tgt - 0.3 * dist : tp > tgt + 0.3 * dist
            n += 1
            planLine(t0, t1, tp, "TP" + str.tostring(k) + " " + str.tostring(tp, format.mintick) + " (" + str.tostring(k) + " R)", cGreen, 1)
    planLine(t0, t1, tgt, "TP" + str.tostring(n + 1) + " = CIBLE " + str.tostring(tgt, format.mintick) + " (" + str.tostring(math.abs(tgt - e) / dist, "#.#") + " R)", cGreen, 2)
    array.push(plL, line.new(t1, e, t1 + H, tgt, xloc = xloc.bar_time, color = c, width = 2, style = line.style_arrow_right))

// Gestion des positions simulées ouvertes
closeSim(int d, float e, float sl, float tp) =>
    risk = math.abs(e - sl)
    float res = na
    hitSL = d == 1 ? low <= sl : high >= sl
    hitTP = d == 1 ? high >= tp : low <= tp
    if hitSL
        res := -1.0
    else if hitTP
        res := math.abs(tp - e) / risk
    else if hUtc >= closeH
        res := (d == 1 ? close - e : e - close) / risk
    na(res) ? na : res - costPts / risk

    r = closeSim(tA, eAp, sAp, gAp)
    if not na(r)
        nA += 1
        wA += r > 0 ? 1 : 0
        rA += r
