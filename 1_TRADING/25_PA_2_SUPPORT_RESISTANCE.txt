//@version=6
// PRICE ACTION 2 — SUPPORTS ET RÉSISTANCES (zones, pas des lignes)
// Méthode :
// 1. Sommets / creux confirmés sur UNE unité de référence FIXE (M15 par défaut) → les MÊMES zones en M1, M5, M10, M15, M30 et H1.
//    En H4 et en jour, le M15 est trop fin : ce sont les sommets / creux de l'unité affichée. (Option : autre référence.)
// 2. Les sommets et creux proches (moins de 0,6 ATR de l'unité de référence) sont regroupés en UNE zone.
//    Une zone n'est gardée que si le prix y a réagi au moins 2 fois (×2, ×3…). Plus il y a de réactions, plus la zone est forte.
// 3. Zone au-dessus du prix = RÉSISTANCE (rouge), en dessous = SUPPORT (vert), prix dedans = ORANGE.
//    Zone TRAVERSÉE (une clôture de chaque côté depuis sa formation) = supprimée : ce n'est plus un support / une résistance.
// 4. On affiche seulement les zones les plus proches (3 au-dessus, 3 en dessous) + plus haut / plus bas de la veille.
//    Le sommet / creux ×1 non traversé le plus proche est aussi affiché (« faible ») s'il est plus près que la 1re zone ×2.
//    S'il n'y a vraiment rien au-dessus (ou en dessous), le prix est au plus haut (plus bas) de la période : le tableau le dit.
// Ne passe aucun ordre.
indicator("Price action 2 — Supports et résistances", shorttitle = "PA S/R", overlay = true, max_boxes_count = 60, max_lines_count = 20, max_labels_count = 60)

srcTf   = input.string("15", "Unité de référence (zones identiques sur toutes les unités jusqu'au H1)", options = ["15", "60", "240", "D", "Unité affichée"])
pivLen  = input.int(3, "Sommet / creux confirmé par N bougies de chaque côté", minval = 2, maxval = 20)
lbDays  = input.int(30, "Historique analysé (jours)", minval = 3, maxval = 365)
nPiv    = input.int(150, "Nombre de sommets / creux analysés", minval = 20, maxval = 300)
tolK    = input.float(0.6, "Largeur max d'une zone (× ATR de la référence)", minval = 0.1, step = 0.1)
minTch  = input.int(2, "Réactions minimum pour garder une zone", minval = 1, maxval = 6)
nSide   = input.int(3, "Zones affichées de chaque côté du prix", minval = 1, maxval = 8)
extH    = input.float(8.0, "Prolonger à droite (heures)", minval = 1, step = 1)
showPD  = input.bool(true, "Plus haut / plus bas de la veille")
showTb  = input.bool(true, "Tableau : support et résistance les plus proches")

cUp = #089981
cDn = #F23645
cIn = #FF9800

refTf = srcTf == "Unité affichée" or timeframe.in_seconds() > 3600 ? timeframe.period : srcTf

// ---------- Sommets / creux sur l'unité de référence ----------
// Sortie : [ATR, puis pour chaque pivot : prix, temps, type (+1 sommet, -1 creux), plus haute clôture depuis, plus basse clôture depuis]
pivData() =>
    hp = ta.pivothigh(high, pivLen, pivLen)
    lp = ta.pivotlow(low, pivLen, pivLen)
    hc = ta.highest(close, pivLen + 1)
    lc = ta.lowest(close, pivLen + 1)
    a = ta.atr(14)
    var array<float> pv = array.new<float>()
    // seulement l'historique récent (calcul rapide, mêmes données sur toutes les unités)
    if time >= timenow - lbDays * 86400000
        // clôtures extrêmes depuis chaque pivot
        if pv.size() > 0
            for i = 0 to int(pv.size() / 5) - 1
                pv.set(i * 5 + 3, math.max(pv.get(i * 5 + 3), close))
                pv.set(i * 5 + 4, math.min(pv.get(i * 5 + 4), close))
        if not na(hp)
            pv.push(hp)
            pv.push(time[pivLen])
            pv.push(1)
            pv.push(hc)
            pv.push(lc)
        if not na(lp)
            pv.push(lp)
            pv.push(time[pivLen])
            pv.push(-1)
            pv.push(hc)
            pv.push(lc)
        while pv.size() > nPiv * 5
            for q = 1 to 5
                pv.shift()
    // la sortie n'est construite que sur la dernière bougie (sinon trop lent en H1 / 30 min)
    out = array.new<float>()
    if barstate.islast
        out.push(a)
        out.concat(pv)
    out

rd = request.security(syminfo.tickerid, refTf, pivData())
[pdh, pdl, dT] = request.security(syminfo.tickerid, "D", [high[1], low[1], time], lookahead = barmerge.lookahead_on)

// ---------- Regroupement en zones ----------
var array<box>   zBx = array.new<box>()
var array<label> zLb = array.new<label>()
var array<line>  pdLn = array.new<line>()
var table tb = table.new(position.bottom_right, 2, 3, border_color = color.new(color.gray, 70), border_width = 1)

fmt(float v) => str.tostring(v, format.mintick)

if barstate.islast
    for b in zBx
        box.delete(b)
    for l in zLb
        label.delete(l)
    for l in pdLn
        line.delete(l)
    zBx.clear()
    zLb.clear()
    pdLn.clear()
    t1 = time + math.round(extH * 3600000)
    // Zones : haut, bas, 1er temps, nombre de réactions, a des sommets, a des creux
    zT = array.new<float>()
    zB = array.new<float>()
    z0 = array.new<int>()
    zN = array.new<int>()
    zH = array.new<bool>()
    zL = array.new<bool>()
    // niveaux ×1 non traversés (secours si aucune zone d'un côté)
    wT = array.new<float>()
    wB = array.new<float>()
    w0 = array.new<int>()
    float atrR = na
    n = na(rd) ? 0 : int((rd.size() - 1) / 5)
    if n > 0
        atrR := rd.get(0)
        tol = tolK * atrR
        pr = array.new<float>()
        for i = 0 to n - 1
            pr.push(rd.get(1 + i * 5))
        idx = pr.sort_indices(order.ascending)
        float cTop = na
        float cBot = na
        int   cT0 = na
        float cMx = na
        float cMn = na
        int   cN = 0
        bool  cH = false
        bool  cL = false
        for k = 0 to n
            bool flush = k == n
            float p = na
            int   tt = na
            float ty = na
            float mx = na
            float mn = na
            if not flush
                j = idx.get(k)
                p := rd.get(1 + j * 5)
                tt := int(rd.get(2 + j * 5))
                ty := rd.get(3 + j * 5)
                mx := rd.get(4 + j * 5)
                mn := rd.get(5 + j * 5)
                flush := cN > 0 and p - cBot > tol
            // zone traversée = clôture au-dessus ET clôture en dessous depuis son 1er pivot → supprimée
            if flush and cN > 0 and not (cMx > cTop and cMn < cBot)
                if cN >= minTch
                    zT.push(cTop)
                    zB.push(cBot)
                    z0.push(cT0)
                    zN.push(cN)
                    zH.push(cH)
                    zL.push(cL)
                else
                    wT.push(cTop)
                    wB.push(cBot)
                    w0.push(cT0)
            if k < n
                if flush or cN == 0
                    cTop := p
                    cBot := p
                    cT0 := tt
                    cMx := mx
                    cMn := mn
                    cN := 1
                    cH := ty > 0
                    cL := ty < 0
                else
                    cTop := math.max(cTop, p)
                    cBot := math.min(cBot, p)
                    if tt < cT0
                        cT0 := tt
                        cMx := mx
                        cMn := mn
                    cN += 1
                    cH := cH or ty > 0
                    cL := cL or ty < 0
    // Sélection : zones les plus proches au-dessus / en dessous, et celles qui contiennent le prix
    m = zT.size()
    float nearR = na
    float nearS = na
    if m > 0
        above = array.new<float>()
        below = array.new<float>()
        for i = 0 to m - 1
            if zB.get(i) > close
                above.push(zB.get(i))
            else if zT.get(i) < close
                below.push(zT.get(i))
        above.sort(order.ascending)
        below.sort(order.descending)
        float limA = above.size() == 0 ? na : above.get(math.min(nSide, above.size()) - 1)
        float limB = below.size() == 0 ? na : below.get(math.min(nSide, below.size()) - 1)
        nearR := above.size() > 0 ? above.get(0) : na
        nearS := below.size() > 0 ? below.get(0) : na
        for i = 0 to m - 1
            top = zT.get(i)
            bot = zB.get(i)
            isR = bot > close
            isS = top < close
            keep = isR ? bot <= limA : isS ? top >= limB : true
            if keep
                // largeur minimale pour rester visible
                if top - bot < 0.15 * atrR
                    mid = (top + bot) / 2
                    top := mid + 0.075 * atrR
                    bot := mid - 0.075 * atrR
                cnt = zN.get(i)
                flip = zH.get(i) and zL.get(i)
                c = isR ? cDn : isS ? cUp : cIn
                tr = cnt >= 4 ? 55 : cnt == 3 ? 70 : 82
                zBx.push(box.new(z0.get(i), top, t1, bot, xloc = xloc.bar_time, border_color = color.new(c, 30), border_width = flip ? 2 : 1, bgcolor = color.new(c, tr)))
                txt = (isR ? "RÉSISTANCE" : isS ? "SUPPORT" : "ZONE (prix dedans)") + " ×" + str.tostring(cnt) + "  " + fmt(bot) + " – " + fmt(top)
                zLb.push(label.new(t1, (top + bot) / 2, txt, xloc = xloc.bar_time, style = label.style_label_left, color = color.new(c, 20), textcolor = color.white, size = size.small))
    // Secours : niveau ×1 le plus proche s'il n'y a aucune zone d'un côté
    bool weakR = false
    bool weakS = false
    if wT.size() > 0
        int iA = na
        int iB = na
        for i = 0 to wT.size() - 1
            if wB.get(i) > close and (na(iA) or wB.get(i) < wB.get(iA))
                iA := i
            if wT.get(i) < close and (na(iB) or wT.get(i) > wT.get(iB))
                iB := i
        for side = 0 to 1
            int ii = side == 0 ? iA : iB
            // affiché s'il n'y a pas de zone ×2 de ce côté, ou si le niveau ×1 est plus proche du prix
            bool need = not na(ii) and (side == 0 ? (na(nearR) or wB.get(ii) < nearR) : (na(nearS) or wT.get(ii) > nearS))
            if need and not na(ii)
                top = wT.get(ii)
                bot = wB.get(ii)
                mid = (top + bot) / 2
                if top - bot < 0.15 * atrR
                    top := mid + 0.075 * atrR
                    bot := mid - 0.075 * atrR
                c = side == 0 ? cDn : cUp
                zBx.push(box.new(w0.get(ii), top, t1, bot, xloc = xloc.bar_time, border_color = color.new(c, 30), border_style = line.style_dashed, bgcolor = color.new(c, 90)))
                zLb.push(label.new(t1, mid, (side == 0 ? "RÉSISTANCE" : "SUPPORT") + " ×1 (faible)  " + fmt(wB.get(ii)) + " – " + fmt(wT.get(ii)), xloc = xloc.bar_time,
                     style = label.style_label_left, color = color.new(c, 50), textcolor = color.white, size = size.small))
                if side == 0
                    nearR := wB.get(ii)
                    weakR := true
                else
                    nearS := wT.get(ii)
                    weakS := true
    // Plus haut / plus bas de la veille
    if showPD and not na(pdh)
        pdLn.push(line.new(dT, pdh, t1, pdh, xloc = xloc.bar_time, color = color.new(color.white, 30), style = line.style_dashed))
        pdLn.push(line.new(dT, pdl, t1, pdl, xloc = xloc.bar_time, color = color.new(color.white, 30), style = line.style_dashed))
        zLb.push(label.new(t1, pdh, "Haut veille " + fmt(pdh), xloc = xloc.bar_time, style = label.style_label_left, color = color.new(color.black, 100), textcolor = color.white, size = size.small))
        zLb.push(label.new(t1, pdl, "Bas veille " + fmt(pdl), xloc = xloc.bar_time, style = label.style_label_left, color = color.new(color.black, 100), textcolor = color.white, size = size.small))
    // Tableau
    table.clear(tb, 0, 0, 1, 2)
    if showTb
        table.cell(tb, 0, 0, "Référence", text_color = color.white, bgcolor = color.new(#131722, 10), text_size = size.small)
        table.cell(tb, 1, 0, refTf == "D" ? "Jour" : refTf == "240" ? "4 h" : refTf == "60" ? "1 h" : refTf + " min", text_color = color.white, bgcolor = color.new(#131722, 10), text_size = size.small)
        table.cell(tb, 0, 1, "Résistance proche", text_color = color.white, bgcolor = color.new(cDn, 40), text_size = size.small)
        table.cell(tb, 1, 1, na(nearR) ? "aucune : prix au plus haut de la période" : fmt(nearR) + "  (+" + str.tostring(nearR - close, "#.##") + ")" + (weakR ? " faible" : ""), text_color = color.white, bgcolor = color.new(cDn, 40), text_size = size.small)
        table.cell(tb, 0, 2, "Support proche", text_color = color.white, bgcolor = color.new(cUp, 40), text_size = size.small)
        table.cell(tb, 1, 2, na(nearS) ? "aucun : prix au plus bas de la période" : fmt(nearS) + "  (−" + str.tostring(close - nearS, "#.##") + ")" + (weakS ? " faible" : ""), text_color = color.white, bgcolor = color.new(cUp, 40), text_size = size.small)
