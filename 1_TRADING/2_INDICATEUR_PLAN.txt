//@version=6
// Plan XAUUSD — checklist achat / vente.
// Claude remplit les réglages (zone, stop, TP de chaque sens) ; l'indicateur dessine la zone,
// le stop et les TP, et coche en direct les 3 conditions d'entrée de chaque sens.
// Fonctionne sur un graphique 1, 5 ou 15 min. Ne passe aucun ordre. Mettre 0 pour désactiver un sens.
indicator("Plan XAUUSD — checklist achat / vente", overlay = true, max_boxes_count = 12, max_lines_count = 12, max_labels_count = 12)

gS = "VENTE"
sZl = input.float(0.0, "Vente : zone bas", group = gS)
sZh = input.float(0.0, "Vente : zone haut", group = gS)
sSl = input.float(0.0, "Vente : stop", group = gS)
sT1 = input.float(0.0, "Vente : TP1", group = gS)
sT2 = input.float(0.0, "Vente : TP2", group = gS)
gB = "ACHAT"
bZl = input.float(0.0, "Achat : zone bas", group = gB)
bZh = input.float(0.0, "Achat : zone haut", group = gB)
bSl = input.float(0.0, "Achat : stop", group = gB)
bT1 = input.float(0.0, "Achat : TP1", group = gB)
bT2 = input.float(0.0, "Achat : TP2", group = gB)
gR = "Réglages"
endHourUtc = input.int(18, "Fin de validité (heure UTC)", minval = 0, maxval = 24, group = gR)
lots       = input.float(0.01, "Taille (lots)", group = gR)
wickRatio  = input.float(0.5, "Mèche minimale (part de la bougie)", minval = 0.1, maxval = 0.9, group = gR)
showDraw   = input.bool(true, "Dessiner zones, stop et TP", group = gR)
tablePos   = input.string("En haut à droite", "Position du tableau", options = ["En haut à droite", "En haut à gauche", "En bas à droite", "En bas à gauche"], group = gR)

sOn = sZl > 0 and sZh > sZl and sSl > sZh
bOn = bZl > 0 and bZh > bZl and bSl > 0 and bSl < bZl

// ---------- Conditions calculées en 15 min (quelle que soit l'unité affichée) ----------
f15() =>
    rng   = high - low
    upW   = high - math.max(open, close)
    loW   = math.min(open, close) - low
    sRej  = high >= sZl and rng > 0 and upW >= wickRatio * rng and close < (sZl + sZh) / 2
    bRej  = low <= bZh and rng > 0 and loW >= wickRatio * rng and close > (bZl + bZh) / 2
    sSin  = ta.barssince(sRej[1])
    bSin  = ta.barssince(bRej[1])
    // Annulé si la dernière clôture au-delà du stop est plus récente que le dernier retour de l'autre côté
    // de la zone (un dépassement ancien, suivi d'un retour sous/au-dessus de la zone, n'annule plus le plan)
    sA    = ta.barssince(close[1] > sSl)
    sB    = ta.barssince(close[1] < sZl)
    bA    = ta.barssince(close[1] < bSl)
    bB    = ta.barssince(close[1] > bZh)
    [ta.highest(high, 4) >= sZl, not na(sSin) and sSin <= 3, not na(sA) and (na(sB) or sA < sB),
     ta.lowest(low, 4) <= bZh, not na(bSin) and bSin <= 3, not na(bA) and (na(bB) or bA < bB)]
[sTouch, sRej, sInv, bTouch, bRej, bInv] = request.security(syminfo.tickerid, "15", f15())

// ---------- Cassure de structure en 5 min (sur bougie clôturée, mémorisée 15 min) ----------
f5() =>
    sBrk = ta.highest(high, 12) >= sZl and close < ta.lowest(low, 6)[1]
    bBrk = ta.lowest(low, 12) <= bZh and close > ta.highest(high, 6)[1]
    sS   = ta.barssince(sBrk[1])
    bS   = ta.barssince(bBrk[1])
    [not na(sS) and sS <= 2, not na(bS) and bS <= 2]
[sBrk, bBrk] = request.security(syminfo.tickerid, "5", f5())

// Heure de la bougie (et non de l'horloge) : fonctionne aussi en mode Replay
expired = hour(time, "UTC") >= endHourUtc
sSig = sOn and sTouch and sRej and sBrk and not sInv and not expired
bSig = bOn and bTouch and bRej and bBrk and not bInv and not expired

// Entrée de référence : clôture de la bougie où le signal apparaît (remise à zéro chaque jour).
// Avant le signal, le R:R est calculé depuis le milieu de la zone.
var float sEntry = na
var float bEntry = na
if ta.change(dayofmonth(time, "UTC")) != 0
    sEntry := na
    bEntry := na
if sSig and na(sEntry)
    sEntry := close
if bSig and na(bEntry)
    bEntry := close
sEp = na(sEntry) ? (sZl + sZh) / 2 : sEntry
bEp = na(bEntry) ? (bZl + bZh) / 2 : bEntry
sRR = sSl - sEp > 0 ? (sEp - sT2) / (sSl - sEp) : na
bRR = bEp - bSl > 0 ? (bT2 - bEp) / (bEp - bSl) : na

alertcondition(sSig, "Signal VENTE XAU", "Plan XAUUSD : les 3 conditions de la VENTE sont réunies")
alertcondition(bSig, "Signal ACHAT XAU", "Plan XAUUSD : les 3 conditions de l'ACHAT sont réunies")

// ---------- Tableau ----------
posMap = tablePos == "En haut à gauche" ? position.top_left : tablePos == "En bas à droite" ? position.bottom_right : tablePos == "En bas à gauche" ? position.bottom_left : position.top_right
var table t = table.new(posMap, 3, 11, bgcolor = color.white, frame_color = color.gray, frame_width = 1, border_color = color.new(color.gray, 80), border_width = 1)

px(float v) => str.tostring(v, "#.##")
mark(bool on, bool c) => not on ? "—" : c ? "✅" : "⬜"
stTxt(bool on, bool inv, bool sig) => not on ? "aucun" : inv ? "ANNULÉ" : expired ? "EXPIRÉ" : sig ? "SIGNAL" : "EN ATTENTE"
rrTxt(bool on, float rr, bool real) => not on or na(rr) ? "—" : str.tostring(rr, "#.#") + (real ? (rr < 1.5 ? " réel : PAS D'ENTRÉE" : " réel") : " (milieu zone)")
rrCol(bool on, float rr) => not on or na(rr) ? color.white : rr >= 1.5 ? color.new(color.green, 70) : color.new(color.red, 60)
stCol(bool on, bool inv, bool sig) => not on or inv or expired ? color.gray : sig ? color.new(color.green, 0) : color.orange

if barstate.isfirst
    table.merge_cells(t, 0, 10, 2, 10)

if barstate.islast
    table.cell(t, 0, 0, "PLAN XAUUSD", text_color = color.white, bgcolor = color.new(color.gray, 20), text_size = size.small)
    table.cell(t, 1, 0, "VENTE", text_color = color.white, bgcolor = color.new(color.red, 15), text_size = size.small)
    table.cell(t, 2, 0, "ACHAT", text_color = color.white, bgcolor = color.new(color.teal, 15), text_size = size.small)
    table.cell(t, 0, 1, "Zone", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 1, sOn ? px(sZl) + "–" + px(sZh) : "—", text_size = size.small)
    table.cell(t, 2, 1, bOn ? px(bZl) + "–" + px(bZh) : "—", text_size = size.small)
    table.cell(t, 0, 2, "1. Prix dans la zone", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 2, mark(sOn, sTouch), text_size = size.small)
    table.cell(t, 2, 2, mark(bOn, bTouch), text_size = size.small)
    table.cell(t, 0, 3, "2. Rejet 15 min (mèche)", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 3, mark(sOn, sRej), text_size = size.small)
    table.cell(t, 2, 3, mark(bOn, bRej), text_size = size.small)
    table.cell(t, 0, 4, "3. Cassure 5 min", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 4, mark(sOn, sBrk), text_size = size.small)
    table.cell(t, 2, 4, mark(bOn, bBrk), text_size = size.small)
    table.cell(t, 0, 5, "Statut", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 5, stTxt(sOn, sInv, sSig), text_color = color.white, bgcolor = stCol(sOn, sInv, sSig), text_size = size.small)
    table.cell(t, 2, 5, stTxt(bOn, bInv, bSig), text_color = color.white, bgcolor = stCol(bOn, bInv, bSig), text_size = size.small)
    table.cell(t, 0, 6, "Stop", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 6, sOn ? px(sSl) : "—", text_size = size.small)
    table.cell(t, 2, 6, bOn ? px(bSl) : "—", text_size = size.small)
    table.cell(t, 0, 7, "TP1 / TP2", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 7, sOn ? px(sT1) + " / " + px(sT2) : "—", text_size = size.small)
    table.cell(t, 2, 7, bOn ? px(bT1) + " / " + px(bT2) : "—", text_size = size.small)
    table.cell(t, 0, 8, "Annulé si", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 8, sOn ? "clôture 15 min > " + px(sSl) : "—", text_size = size.small)
    table.cell(t, 2, 8, bOn ? "clôture 15 min < " + px(bSl) : "—", text_size = size.small)
    table.cell(t, 0, 9, "R:R au TP2", text_halign = text.align_left, text_size = size.small)
    table.cell(t, 1, 9, rrTxt(sOn, sRR, not na(sEntry)), bgcolor = rrCol(sOn, sRR), text_size = size.small)
    table.cell(t, 2, 9, rrTxt(bOn, bRR, not na(bEntry)), bgcolor = rrCol(bOn, bRR), text_size = size.small)
    table.cell(t, 0, 10, "Taille " + str.tostring(lots) + " lot — stop et TP activés — expire à " + str.tostring(endHourUtc) + ":00 UTC — annulé aussi si le TP2 est atteint avant l'entrée — à 0,01 lot : stop à l'entrée à 1R, sortie au TP2", text_halign = text.align_left, text_size = size.tiny)

// ---------- Dessins : zone d'entrée (à gauche) + outil « position » (à droite) ----------
// Même lecture que l'outil Position longue / courte de TradingView : rouge = risque (entrée → stop),
// vert = objectif (entrée → TP2). L'entrée est le milieu de la zone, puis le prix réel au signal.
var array<box>   bxs = array.new<box>()
var array<line>  lns = array.new<line>()
var array<label> lbs = array.new<label>()

drawPlan(string side, color zc, float zl, float zh, float ep, float sl, float t1, float t2, float rr, bool real, string invTxt) =>
    array.push(bxs, box.new(bar_index - 40, zh, bar_index, zl, border_color = zc, bgcolor = color.new(zc, 85), text = "ZONE " + side, text_color = zc, text_size = size.small))
    array.push(bxs, box.new(bar_index, math.max(ep, sl), bar_index + 30, math.min(ep, sl), border_color = color.red, bgcolor = color.new(color.red, 75), text = "SL " + str.tostring(sl, "#.##") + " — " + invTxt, text_color = color.red, text_size = size.small))
    array.push(bxs, box.new(bar_index, math.max(ep, t2), bar_index + 30, math.min(ep, t2), border_color = color.green, bgcolor = color.new(color.green, 80), text = "TP2 " + str.tostring(t2, "#.##"), text_color = color.green, text_size = size.small))
    array.push(lns, line.new(bar_index, ep, bar_index + 30, ep, color = color.gray, width = 2))
    array.push(lns, line.new(bar_index, t1, bar_index + 30, t1, color = color.green, style = line.style_dashed))
    array.push(lbs, label.new(bar_index + 30, ep, side + (real ? " entrée réelle " : " entrée (milieu zone) ") + str.tostring(ep, "#.##") + " — R:R " + (na(rr) ? "—" : str.tostring(rr, "#.#")), style = label.style_label_left, color = color.new(color.gray, 20), textcolor = color.white, size = size.small))
    array.push(lbs, label.new(bar_index + 30, t1, "TP1 " + str.tostring(t1, "#.##"), style = label.style_label_left, color = color.new(color.green, 85), textcolor = color.green, size = size.small))

if barstate.islast
    for b in bxs
        box.delete(b)
    for l in lns
        line.delete(l)
    for lb in lbs
        label.delete(lb)
    array.clear(bxs)
    array.clear(lns)
    array.clear(lbs)
    if showDraw and sOn
        drawPlan("VENTE", color.red, sZl, sZh, sEp, sSl, sT1, sT2, sRR, not na(sEntry), "annulé si clôture 15 min au-dessus")
    if showDraw and bOn
        drawPlan("ACHAT", color.teal, bZl, bZh, bEp, bSl, bT1, bT2, bRR, not na(bEntry), "annulé si clôture 15 min en dessous")
