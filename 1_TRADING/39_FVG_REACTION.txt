//@version=6
// FVG RÉACTION — retour sur un FVG + bougie de confirmation (méthode de Farid). Code original.
// Règles (version ACHAT sur un FVG haussier ; la VENTE est l'inverse) :
//  1. FVG haussier = trou entre le haut de la bougie A et le bas de la bougie C (taille ≥ 0,3 ATR par défaut).
//  2. BOUGIE 1 = la 1re bougie qui revient TOUCHER le FVG (1er retour seulement).
//     Si elle clôture SOUS le FVG, le FVG est cassé : pas de trade.
//  3. BOUGIE 2 = la bougie suivante. Elle doit CLÔTURER AU-DESSUS DU HAUT de la bougie 1. Sinon : pas de trade.
//  4. ENTRÉE à la clôture de la bougie 2. STOP sous le plus bas des bougies 1 et 2. OBJECTIF 2 R.
//  5. Tableau : résultats en R, frais compris (stop compté en premier si stop et objectif dans la même bougie).
// Filtres en option : tendance H1 / H4, killzones (Londres 9h–11h, New York 14h30–17h, Paris). Ne passe aucun ordre.
indicator("FVG : retour + bougie de confirmation", shorttitle = "FVG RÉACTION", overlay = true,
     max_boxes_count = 300, max_lines_count = 400, max_labels_count = 300, max_bars_back = 500)

gF = "FVG"
minK   = input.float(0.3, "Taille minimum du FVG (× ATR 14)", minval = 0, step = 0.05, group = gF)
maxAge = input.int(100, "FVG abandonné s'il n'est pas retouché après (bougies)", minval = 5, maxval = 500, group = gF)
keepN  = input.int(8, "FVG en attente gardés par sens", minval = 1, maxval = 30, group = gF)
gT = "Trade"
rr      = input.float(2.0, "Objectif (en R)", minval = 0.5, step = 0.5, group = gT)
bufK    = input.float(0.0, "Marge du stop sous / au-dessus des bougies 1 et 2 (× ATR 14)", minval = 0, step = 0.05, group = gT)
maxHold = input.int(96, "Sortie au temps après (bougies)", minval = 5, maxval = 400, group = gT)
costPts = input.float(0.37, "Frais aller-retour en prix (or 0,37 ; EURUSD 0,00012)", minval = 0, step = 0.01, group = gT)
gX = "Filtres"
trendMode = input.string("Aucun", "Tendance de fond", options = ["Aucun", "H1 (EMA 50)", "H4 (EMA 50)"], group = gX)
sessOn  = input.bool(false, "Signaux seulement en killzone (Londres 9h–11h, New York 14h30–17h, Paris)", group = gX)
dirMode = input.string("Achats et ventes", "Sens", options = ["Achats et ventes", "Achats seulement", "Ventes seulement"], group = gX)
gA = "Affichage"
showOld = input.bool(false, "Garder les FVG terminés (grisés)", group = gA)
showTab = input.bool(true, "Afficher le tableau des résultats", group = gA)

cUp = #089981
cDn = #F23645
cGr = color.gray

atr = ta.atr(14)
fmt(float v) => str.tostring(v, format.mintick)
kz = not na(time(timeframe.period, "0900-1100,1430-1700", "Europe/Paris"))
h1Bull = request.security(syminfo.tickerid, "60", close[1] > ta.ema(close, 50)[1], lookahead = barmerge.lookahead_on)
h4Bull = request.security(syminfo.tickerid, "240", close[1] > ta.ema(close, 50)[1], lookahead = barmerge.lookahead_on)
filtOk(int d) =>
    bool ok = dirMode == "Achats et ventes" or (dirMode == "Achats seulement" and d == 1) or (dirMode == "Ventes seulement" and d == -1)
    if trendMode == "H1 (EMA 50)"
        ok := ok and (d == 1 ? h1Bull : not h1Bull)
    else if trendMode == "H4 (EMA 50)"
        ok := ok and (d == 1 ? h4Bull : not h4Bull)
    ok and (not sessOn or kz)

type Fvg
    int   dir
    float top
    float bot
    int   born
    box   bx
    int   st = 0
    int   c1Bar = na
    float c1H = na
    float c1L = na
    bool  traded = false

type Trd
    int   dir
    float e
    float sl
    float tp
    int   b
    label lb
    line  le
    line  ls
    line  lt

var array<Fvg> fvgs = array.new<Fvg>()
var array<Trd> trs = array.new<Trd>()
var int   nTr = 0
var int   nW = 0
var float sumR = 0.0
var float gW = 0.0
var float gL = 0.0
var int   nTouch = 0
bool newSig = false

// ---------- 1. Trades en cours : stop / objectif / temps ----------
if trs.size() > 0
    for i = trs.size() - 1 to 0
        t = trs.get(i)
        if bar_index > t.b
            rk = math.abs(t.e - t.sl)
            hitS = t.dir == 1 ? low <= t.sl : high >= t.sl
            hitT = t.dir == 1 ? high >= t.tp : low <= t.tp
            float res = na
            if hitS
                res := -1.0
            else if hitT
                res := rr
            else if bar_index - t.b >= maxHold
                res := t.dir * (close - t.e) / rk
            line.set_x2(t.le, bar_index)
            line.set_x2(t.ls, bar_index)
            line.set_x2(t.lt, bar_index)
            if not na(res)
                res -= costPts / rk
                nTr += 1
                sumR += res
                if res > 0
                    nW += 1
                    gW += res
                else
                    gL -= res
                label.set_text(t.lb, label.get_text(t.lb) + (hitS ? "\n✗ stop " : hitT ? "\n✓ objectif " : "\n⏱ temps ") + str.tostring(res, "+#.##;-#.##") + " R")
                if hitS
                    label.set_color(t.lb, color.new(cGr, 20))
                trs.remove(i)

// ---------- 2. FVG en attente : bougie 1 (contact), puis bougie 2 (confirmation) ----------
if fvgs.size() > 0
    for i = fvgs.size() - 1 to 0
        f = fvgs.get(i)
        if f.born < bar_index
            if f.st == 1
                // bougie 2 = la bougie juste après le contact
                conf = f.dir == 1 ? close > f.c1H : close < f.c1L
                f.st := 2
                if conf and filtOk(f.dir)
                    e = close
                    sl = f.dir == 1 ? math.min(f.c1L, low) - bufK * atr : math.max(f.c1H, high) + bufK * atr
                    rk = math.abs(e - sl)
                    if rk > 0
                        tp = e + f.dir * rr * rk
                        c = f.dir == 1 ? cUp : cDn
                        lb = label.new(bar_index, f.dir == 1 ? low : high, (f.dir == 1 ? "ACHAT " : "VENTE ") + fmt(e) + "\nSL " + fmt(sl) + " | TP " + fmt(tp),
                             style = f.dir == 1 ? label.style_label_up : label.style_label_down, color = c, textcolor = color.white, size = size.tiny)
                        le = line.new(bar_index, e, bar_index + 1, e, color = c, width = 2)
                        ls = line.new(bar_index, sl, bar_index + 1, sl, color = cDn, style = line.style_dotted)
                        lt = line.new(bar_index, tp, bar_index + 1, tp, color = cUp, style = line.style_dotted)
                        trs.push(Trd.new(f.dir, e, sl, tp, bar_index, lb, le, ls, lt))
                        newSig := true
                        box.set_bgcolor(f.bx, color.new(c, 70))
                        f.traded := true
            else if f.st == 0
                touched = f.dir == 1 ? low <= f.top : high >= f.bot
                if touched
                    nTouch += 1
                    broke = f.dir == 1 ? close < f.bot : close > f.top
                    if broke
                        f.st := 2
                    else
                        f.st := 1
                        f.c1Bar := bar_index
                        f.c1H := high
                        f.c1L := low
                    box.set_right(f.bx, bar_index)
                else if bar_index - f.born > maxAge
                    f.st := 2
                else
                    box.set_right(f.bx, bar_index + 3)
            if f.st == 2
                // FVG tradé : gardé en couleur ; sinon grisé (option) ou effacé
                if f.traded
                    box.set_right(f.bx, bar_index)
                else if showOld
                    box.set_right(f.bx, bar_index)
                    box.set_bgcolor(f.bx, color.new(cGr, 88))
                    box.set_border_color(f.bx, color.new(cGr, 60))
                else
                    box.delete(f.bx)
                fvgs.remove(i)

// ---------- 3. Nouveau FVG ----------
if bar_index > 2
    if low > high[2] and low - high[2] >= minK * atr
        fvgs.push(Fvg.new(1, low, high[2], bar_index, box.new(bar_index - 2, low, bar_index + 3, high[2], border_color = color.new(cUp, 40),
             bgcolor = color.new(cUp, 85), text = "FVG", text_color = cUp, text_size = size.tiny, text_halign = text.align_right)))
    if high < low[2] and low[2] - high >= minK * atr
        fvgs.push(Fvg.new(-1, low[2], high, bar_index, box.new(bar_index - 2, low[2], bar_index + 3, high, border_color = color.new(cDn, 40),
             bgcolor = color.new(cDn, 85), text = "FVG", text_color = cDn, text_size = size.tiny, text_halign = text.align_right)))
    // garder les keepN FVG en attente les plus récents par sens
    for d in array.from(1, -1)
        int cnt = 0
        for f in fvgs
            cnt += f.dir == d and f.st == 0 ? 1 : 0
        if cnt > keepN
            for i = 0 to fvgs.size() - 1
                f = fvgs.get(i)
                if f.dir == d and f.st == 0
                    box.delete(f.bx)
                    fvgs.remove(i)
                    break

// ---------- Tableau ----------
var table tb = table.new(position.bottom_right, 2, 7, bgcolor = color.new(#131722, 10), border_width = 1, border_color = color.new(cGr, 70))
if barstate.islast and showTab
    pf = gL > 0 ? gW / gL : na
    avgR = nTr > 0 ? sumR / nTr : na
    table.cell(tb, 0, 0, "FVG RÉACTION", text_color = color.white, text_size = size.small)
    table.cell(tb, 1, 0, str.tostring(rr) + " R · " + trendMode, text_color = color.silver, text_size = size.small)
    table.cell(tb, 0, 1, "FVG retouchés → trades", text_color = color.silver, text_size = size.small)
    table.cell(tb, 1, 1, str.tostring(nTouch) + " → " + str.tostring(nTr + trs.size()), text_color = color.white, text_size = size.small)
    table.cell(tb, 0, 2, "Gagnants", text_color = color.silver, text_size = size.small)
    table.cell(tb, 1, 2, nTr > 0 ? str.tostring(100.0 * nW / nTr, "#") + " %" : "—", text_color = color.white, text_size = size.small)
    table.cell(tb, 0, 3, "R moyen / trade", text_color = color.silver, text_size = size.small)
    table.cell(tb, 1, 3, na(avgR) ? "—" : str.tostring(avgR, "+#.##;-#.##") + " R", text_color = na(avgR) ? color.white : avgR > 0 ? cUp : cDn, text_size = size.small)
    table.cell(tb, 0, 4, "Total", text_color = color.silver, text_size = size.small)
    table.cell(tb, 1, 4, str.tostring(sumR, "+#.#;-#.#") + " R", text_color = sumR > 0 ? cUp : cDn, text_size = size.small)
    table.cell(tb, 0, 5, "PF (en R)", text_color = color.silver, text_size = size.small)
    table.cell(tb, 1, 5, na(pf) ? "—" : str.tostring(pf, "#.##"), text_color = color.white, text_size = size.small)
    table.cell(tb, 0, 6, "Seuil à " + str.tostring(rr) + " R", text_color = color.silver, text_size = size.small)
    table.cell(tb, 1, 6, str.tostring(100.0 / (1 + rr), "#") + " % + frais", text_color = cGr, text_size = size.small)

alertcondition(newSig, "FVG réaction : entrée", "Retour sur FVG + bougie de confirmation : entrée à la clôture")
