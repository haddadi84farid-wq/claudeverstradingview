//@version=6
// ORDER BLOCKS MULTI-UNITÉS — la méthode de la vidéo, étape par étape, IDENTIQUE quelle que soit l'unité affichée.
// 1. Les OB sont détectés sur 5, 10, 15, 30 min et 1 h (+ 1 min en option) avec la même règle :
//    cassure de la structure EXTERNE (vrai sommet / vrai creux, pas un petit pivot interne) → OB = dernière bougie (ou série)
//    opposée juste avant la jambe qui casse. Filtres OB « institutionnel » : déplacement ≥ 1,5 ATR + FVG obligatoire.
//    On garde les 3 derniers OB vendeurs et acheteurs de chaque unité, encore valables.
// 2. On ne garde que les OB qui se SUPERPOSENT sur plusieurs unités (« ×3 », « ×4 »…) : ce sont les niveaux forts.
// 3. Étoiles sur l'OB le plus précis du groupe (la plus petite unité) : F = FVG, T = Supertrend, P = premium / discount
//    (Fibonacci du dernier sommet au plus bas atteint, OB vendeur au-dessus de 50 %), V = vierge, M = multi-unités (≥ 3),
//    H = le 1 h est dans le groupe (zone la plus fiable). Note sur 6.
// 4. Fibonacci DANS l'OB (0 / 0,5 / 1) : entrée sur la 2e moitié possible, stop juste au-delà de l'OB, objectif 2 R.
// 5. Tableau des résultats : chaque groupe (×2 et plus) est testé sur l'historique avec un ordre limite à l'entrée,
//    stop et objectif du plan, frais inclus, annulé s'il n'est pas touché à temps. Résultats en R par catégorie.
// Ne passe aucun ordre.
indicator("Order Blocks multi-unités", shorttitle = "OB MTF", overlay = true, max_boxes_count = 100, max_lines_count = 150, max_labels_count = 100, max_bars_back = 500)

// ---------- Détection sur UNE unité (exécutée dans chaque unité via request.security) ----------
type OB
    int   dir
    float top
    float bot
    int   t
    int   c
    bool  f
    bool  tr
    bool  v
    float w0
    float ex
    float atr0

obTF() =>
    ph = ta.pivothigh(high, pivLen, pivLen)
    pl = ta.pivotlow(low, pivLen, pivLen)
    [stV, stD] = ta.supertrend(stFac, stLen)
    atr = ta.atr(14)
    var float sh = na
    var float sl = na
    var int   shBar = na
    var int   slBar = na
    var bool  shB = true
    var bool  slB = true
    var array<OB> obs = array.new<OB>()
    // Structure : en mode externe, un pivot plus bas que le sommet protégé (non cassé) est interne → ignoré
    if not na(ph) and (not extOnly or shB or na(sh) or ph > sh)
        sh := ph
        shBar := bar_index - pivLen
        shB := false
    if not na(pl) and (not extOnly or slB or na(sl) or pl < sl)
        sl := pl
        slBar := bar_index - pivLen
        slB := false
    lenL = math.min(300, math.max(1, bar_index - nz(shBar, bar_index) + 1))
    lenH = math.min(300, math.max(1, bar_index - nz(slBar, bar_index) + 1))
    llS = ta.lowest(low, lenL)
    llO = ta.lowestbars(low, lenL)
    hhS = ta.highest(high, lenH)
    hhO = ta.highestbars(high, lenH)
    // Suivi : stop du plan touché (mèche au-delà de l'OB + marge) → OB mort, supprimé ; touché → plus vierge ;
    // FVG peut se former dans les 2 bougies après la cassure
    if obs.size() > 0
        for i = obs.size() - 1 to 0
            o = obs.get(i)
            if o.dir == -1
                if high > o.top + slBuf
                    obs.remove(i)
                else
                    if not o.f and bar_index - o.c <= 2 and high < low[2]
                        o.f := true
                    if bar_index > o.c and high >= o.bot
                        o.v := false
                    else if o.v
                        o.ex := math.min(o.ex, low)
            else
                if low < o.bot - slBuf
                    obs.remove(i)
                else
                    if not o.f and bar_index - o.c <= 2 and low > high[2]
                        o.f := true
                    if bar_index > o.c and low <= o.top
                        o.v := false
                    else if o.v
                        o.ex := math.max(o.ex, high)
    // Cassure vers le bas → OB vendeur
    if not slB and not na(sl) and close < sl
        slB := true
        int j = 1
        for k = 1 to 20
            if close[k] > open[k]
                j := k
                break
        int j2 = j
        for k = j + 1 to j + 4
            if close[k] > open[k]
                j2 := k
            else
                break
        float tp = high[j]
        float bt = low[j]
        for k = j to j2
            tp := math.max(tp, high[k])
            bt := math.min(bt, low[k])
        float ex = low
        for k = 0 to j - 1
            ex := math.min(ex, low[k])
        bool f = false
        if j >= 2
            for k = 0 to j - 2
                if high[k] < low[k + 2]
                    f := true
        obs.push(OB.new(-1, tp, bt, time[j2], bar_index, f, stD > 0, true, math.max(hhS, tp), ex, atr))
        if extOnly
            // nouveau sommet protégé = haut de la jambe qui a cassé
            sh := hhS
            shBar := bar_index + hhO
            shB := false
    // Cassure vers le haut → OB acheteur
    if not shB and not na(sh) and close > sh
        shB := true
        int j = 1
        for k = 1 to 20
            if close[k] < open[k]
                j := k
                break
        int j2 = j
        for k = j + 1 to j + 4
            if close[k] < open[k]
                j2 := k
            else
                break
        float tp = high[j]
        float bt = low[j]
        for k = j to j2
            tp := math.max(tp, high[k])
            bt := math.min(bt, low[k])
        float ex = high
        for k = 0 to j - 1
            ex := math.max(ex, high[k])
        bool f = false
        if j >= 2
            for k = 0 to j - 2
                if low[k] > high[k + 2]
                    f := true
        obs.push(OB.new(1, tp, bt, time[j2], bar_index, f, stD < 0, true, math.min(llS, bt), ex, atr))
        if extOnly
            sl := llS
            slBar := bar_index + llO
            slB := false
    // On garde les keepN derniers par sens
    for d in array.from(-1, 1)
        int cnt = 0
        for o in obs
            if o.dir == d
                cnt += 1
        while cnt > keepN
            for i = 0 to obs.size() - 1
                if obs.get(i).dir == d
                    obs.remove(i)
                    break
            cnt -= 1
    // Sortie : 9 valeurs par OB → sens, haut, bas, temps, F, T, P, V, déplacement OK
    out = array.new<float>()
    for o in obs
        mid = (o.top + o.bot) / 2
        p = o.dir == -1 ? mid >= (o.w0 + o.ex) / 2 : mid <= (o.w0 + o.ex) / 2
        dOk = (o.dir == -1 ? o.top - o.ex : o.ex - o.bot) >= dispK * o.atr0
        out.push(o.dir)
        out.push(o.top)
        out.push(o.bot)
        out.push(o.t)
        out.push(o.f ? 1 : 0)
        out.push(o.tr ? 1 : 0)
        out.push(p ? 1 : 0)
        out.push(o.v ? 1 : 0)
        out.push(dOk ? 1 : 0)
    out

r1 = request.security(syminfo.tickerid, "1", obTF())

// Ajoute les OB valides d'une unité (sens d) dans les listes
collect(array<float> r, int tfi, int d, array<float> tps, array<float> bts, array<int> ts, array<int> tfs, array<int> fs, array<int> rs, array<int> ps, array<int> vs) =>
    n = na(r) ? 0 : int(r.size() / 9)
    if n > 0
        for i = 0 to n - 1
            b = i * 9
            if r.get(b) == d and r.get(b + 8) == 1 and (not reqF or r.get(b + 4) == 1)
                tps.push(r.get(b + 1))
                bts.push(r.get(b + 2))
                ts.push(int(r.get(b + 3)))
                tfs.push(tfi)
                fs.push(int(r.get(b + 4)))
                rs.push(int(r.get(b + 5)))
                ps.push(int(r.get(b + 6)))
                vs.push(int(r.get(b + 7)))
    n

mkLists(int d) =>
    tps = array.new<float>()
    bts = array.new<float>()
    ts  = array.new<int>()
    tfs = array.new<int>()
    fs  = array.new<int>()
    rs  = array.new<int>()
    ps  = array.new<int>()
    vs  = array.new<int>()
    if with1
        collect(r1, 0, d, tps, bts, ts, tfs, fs, rs, ps, vs)
    collect(r5, 1, d, tps, bts, ts, tfs, fs, rs, ps, vs)
    collect(rA, 2, d, tps, bts, ts, tfs, fs, rs, ps, vs)
    collect(rB, 3, d, tps, bts, ts, tfs, fs, rs, ps, vs)
    collect(rC, 4, d, tps, bts, ts, tfs, fs, rs, ps, vs)
    collect(rD, 5, d, tps, bts, ts, tfs, fs, rs, ps, vs)
    [tps, bts, ts, tfs, fs, rs, ps, vs]

// Groupes (×2 et plus) : indice de l'OB le plus précis, nombre d'unités, noms, présence du 1 h
grp(array<float> tps, array<float> bts, array<int> tfs) =>
    n = tps.size()
    gI = array.new<int>()
    gN = array.new<int>()
    gS = array.new<string>()
    gH = array.new<bool>()
    if n > 0
        used = array.new<bool>(n, false)
        // Les OB sont rangés de la plus petite à la plus grande unité : le 1er d'un groupe est le plus précis
        for i = 0 to n - 1
            if not used.get(i)
                ti = tps.get(i)
                bi = bts.get(i)
                seen = array.new<bool>(6, false)
                seen.set(tfs.get(i), true)
                for k = 0 to n - 1
                    if k != i and not used.get(k) and math.min(ti, tps.get(k)) > math.max(bi, bts.get(k))
                        seen.set(tfs.get(k), true)
                int cnt = 0
                string names = ""
                for q = 0 to 5
                    if seen.get(q)
                        cnt += 1
                        names := names == "" ? tfNames.get(q) : names + "·" + tfNames.get(q)
                if cnt >= 2
                    for k = 0 to n - 1
                        if not used.get(k) and math.min(ti, tps.get(k)) > math.max(bi, bts.get(k))
                            used.set(k, true)
                    gI.push(i)
                    gN.push(cnt)
                    gS.push(names)
                    gH.push(seen.get(5))
    [gI, gN, gS, gH]

planOf(int d, float ti, float bi) =>
    e = entryAt == "Bord de l'OB" ? (d == -1 ? bi : ti) : (ti + bi) / 2
    slp = d == -1 ? ti + slBuf : bi - slBuf
    tpp = e + (d == -1 ? -1 : 1) * rrTP * math.abs(e - slp)
    [e, slp, tpp]

[sTp, sBt, sTs, sTf, sF, sR, sP, sV] = mkLists(-1)
[uTp, uBt, uTs, uTf, uF, uR, uP, uV] = mkLists(1)
[sgI, sgN, sgS, sgH] = grp(sTp, sBt, sTf)
[ugI, ugN, ugS, ugH] = grp(uTp, uBt, uTf)

// ---------- Test historique : ordre limite sur chaque nouveau groupe ----------
type Setup
    int   dir
    float e
    float sl
    float tp
    int   st
    int   cnt
    bool  h1
    int   t0
    int   c
    bool  filled

var array<Setup>  sets = array.new<Setup>()
var array<string> keys = array.new<string>()
// Catégories : 0 note 5-6, 1 note 4, 2 note ≤ 3, 3 avec 1 h, 4 ×3 et plus, 5 ×2 sans 1 h, 6 tous
var array<int>   stN = array.new<int>(7, 0)
var array<int>   stW = array.new<int>(7, 0)
var array<float> stR = array.new<float>(7, 0.0)

addRes(int cat, float r) =>
    stN.set(cat, stN.get(cat) + 1)
    stW.set(cat, stW.get(cat) + (r > 0 ? 1 : 0))
    stR.set(cat, stR.get(cat) + r)
    cat

regGroups(int d, array<float> tps, array<float> bts, array<int> ts, array<int> fs, array<int> rs, array<int> ps, array<int> vs, array<int> gI, array<int> gN, array<bool> gH) =>
    if gI.size() > 0
        for g = 0 to gI.size() - 1
            i = gI.get(g)
            ti = tps.get(i)
            bi = bts.get(i)
            key = str.tostring(d) + "|" + str.tostring(ts.get(i)) + "|" + str.tostring(ti)
            if not keys.includes(key)
                keys.push(key)
                if keys.size() > 400
                    keys.shift()
                [e, slp, tpp] = planOf(d, ti, bi)
                // ordre limite seulement si le prix n'a pas déjà dépassé l'entrée
                if (d == 1 ? close > e : close < e) and math.abs(e - slp) > 0
                    cnt = gN.get(g)
                    h1 = gH.get(g)
                    st = fs.get(i) + rs.get(i) + ps.get(i) + vs.get(i) + (cnt >= 3 ? 1 : 0) + (h1 ? 1 : 0)
                    sets.push(Setup.new(d, e, slp, tpp, st, cnt, h1, time, bar_index, false))
    d

// Suivi des ordres en cours (les ordres créés sur cette bougie attendent la suivante)
if sets.size() > 0
    for i = sets.size() - 1 to 0
        s = sets.get(i)
        if bar_index > s.c
            bool drop = false
            float res = na
            if not s.filled
                if s.dir == 1 ? low <= s.e : high >= s.e
                    s.filled := true
                else if time - s.t0 > expH * 3600000
                    drop := true
            if s.filled
                cst = costPts / math.abs(s.e - s.sl)
                if s.dir == 1 ? low <= s.sl : high >= s.sl
                    res := -1 - cst
                else if s.dir == 1 ? high >= s.tp : low <= s.tp
                    res := rrTP - cst
            if not na(res)
                addRes(s.st >= 5 ? 0 : s.st == 4 ? 1 : 2, res)
                if s.h1
                    addRes(3, res)
                if s.cnt >= 3
                    addRes(4, res)
                if s.cnt == 2 and not s.h1
                    addRes(5, res)
                addRes(6, res)
            if drop or not na(res)
                sets.remove(i)

regGroups(-1, sTp, sBt, sTs, sF, sR, sP, sV, sgI, sgN, sgH)
regGroups(1, uTp, uBt, uTs, uF, uR, uP, uV, ugI, ugN, ugH)

// ---------- Dessin (dernière bougie) ----------
drawGroups(int d, array<float> tps, array<float> bts, array<int> ts, array<int> fs, array<int> rs, array<int> ps, array<int> vs, array<int> gI, array<int> gN, array<string> gS, array<bool> gH) =>
    if gI.size() > 0
        for g = 0 to gI.size() - 1
            i = gI.get(g)
            cnt = gN.get(g)
            h1 = gH.get(g)
            if cnt >= minConf or (h1Pass and h1 and cnt >= 2)
                ti = tps.get(i)
                bi = bts.get(i)
                c = d == -1 ? cRed : cGreen
                st = fs.get(i) + rs.get(i) + ps.get(i) + vs.get(i) + (cnt >= 3 ? 1 : 0) + (h1 ? 1 : 0)
                strong = h1 or st >= 5
                t0 = ts.get(i)
                t1 = time + math.round(extH * 3600000)
                midp = (ti + bi) / 2
                [e, slp, tpp] = planOf(d, ti, bi)
                gBx.push(box.new(t0, ti, t1, bi, xloc = xloc.bar_time, border_color = c, border_width = strong ? 2 : 1,
                     bgcolor = color.new(c, strong ? 70 : 85)))
                // Fibonacci dans l'OB : 0 / 0,5 / 1
                gLn.push(line.new(t0, midp, t1, midp, xloc = xloc.bar_time, color = color.new(c, 20), style = line.style_dashed))
                txt = (d == -1 ? "OB VENTE ×" : "OB ACHAT ×") + str.tostring(cnt) + " (" + gS.get(g) + ")  " + str.tostring(st) + "/6 " + starTxt.get(st) +
                     "\nF" + ck(fs.get(i) == 1) + " T" + ck(rs.get(i) == 1) + " P" + ck(ps.get(i) == 1) + " V" + ck(vs.get(i) == 1) + " M" + ck(cnt >= 3) + " H" + ck(h1) +
                     "\nEntrée " + str.tostring(e, format.mintick) + " | SL " + str.tostring(slp, format.mintick) + " | TP " + str.tostring(tpp, format.mintick) + " (" + str.tostring(rrTP) + " R)"
                gLb.push(label.new(t1, d == -1 ? ti : bi, txt, xloc = xloc.bar_time, style = d == -1 ? label.style_label_lower_left : label.style_label_upper_left,
                     color = color.new(c, strong ? 0 : 35), textcolor = color.white, size = size.small))
    d
