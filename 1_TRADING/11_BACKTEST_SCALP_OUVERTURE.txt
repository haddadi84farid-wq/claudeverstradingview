//@version=6
// Backtest SCALPING 5 min : cassure de la 1re demi-heure de Londres et de New York,
// UNIQUEMENT dans le sens de la tendance journalière (cassure 20 jours / sortie 10 jours, valeur de la veille).
// À mettre sur XAUUSD en 5 MIN. Frais inclus : glissement 15 ticks par ordre (≈ spread + glissement)
// et commission 0,035 $ par once et par ordre (≈ 7 $ par lot aller-retour, compte Razor).
// Règles :
// - Range d'ouverture : plus haut / plus bas des 30 premières minutes de la séance (Londres 07:00 UTC, New York 13:30 UTC en été ;
//   en hiver, mettre 08:00 et 14:30).
// - Entrée : première clôture 5 min au-delà du range, dans le sens de la tendance journalière, avant l'heure limite.
// - Stop : milieu du range (ou l'autre bord, au choix), au moins « stop min » × ATR 5 min. TP : « TP en R ». Stop au point d'entrée à 1 R.
// - 1 trade maximum par séance ; tout est fermé à l'heure de fin de séance.
strategy("Backtest XAU — scalping ouverture", overlay = true, initial_capital = 10000,
     default_qty_type = strategy.fixed, default_qty_value = 1, slippage = 15, pyramiding = 0,
     margin_long = 1, margin_short = 1, commission_type = strategy.commission.cash_per_contract, commission_value = 0.035)

riskCash  = input.float(100.0, "Risque par trade ($)", minval = 1)
useTrend  = input.bool(true, "Filtre tendance journalière (seulement dans son sens)")
stopMid   = input.bool(true, "Stop au milieu du range (sinon à l'autre bord)")
minStopA  = input.float(1.0, "Stop minimum (× ATR 5 min)", minval = 0.2)
tpR       = input.float(1.5, "TP en R", minval = 0.5)
beR       = input.float(1.0, "Stop au point d'entrée à (R)", minval = 0.3)
useLon    = input.bool(true, "Séance de Londres")
lonH      = input.int(7, "Londres : heure d'ouverture (UTC)", minval = 0, maxval = 23)
lonEnd    = input.int(10, "Londres : entrées jusqu'à (heure UTC)", minval = 1, maxval = 23)
lonFlat   = input.int(12, "Londres : clôture forcée (heure UTC)", minval = 1, maxval = 23)
useNy     = input.bool(true, "Séance de New York")
nyH       = input.int(13, "New York : heure d'ouverture (UTC)", minval = 0, maxval = 23)
nyM       = input.int(30, "New York : minute d'ouverture", minval = 0, maxval = 59)
nyEnd     = input.int(16, "New York : entrées jusqu'à (heure UTC)", minval = 1, maxval = 23)
nyFlat    = input.int(19, "New York : clôture forcée (heure UTC)", minval = 1, maxval = 23)

atr = ta.atr(14)
h   = hour(time, "UTC")
mi  = minute(time, "UTC")
tod = h * 60 + mi

// Tendance journalière de la veille : 1 achat, -1 vente, 0 neutre
dayTrend() =>
    var int st = 0
    if close > ta.highest(high, 20)[1]
        st := 1
    else if close < ta.lowest(low, 20)[1]
        st := -1
    else if st == 1 and close < ta.lowest(low, 10)[1]
        st := 0
    else if st == -1 and close > ta.highest(high, 10)[1]
        st := 0
    st
dTrend = request.security(syminfo.tickerid, "D", dayTrend()[1], lookahead = barmerge.lookahead_on)

// Séance en cours : 1 = Londres, 2 = New York
lonStart = lonH * 60
nyStart  = nyH * 60 + nyM
inLonRange = useLon and tod >= lonStart and tod < lonStart + 30
inNyRange  = useNy and tod >= nyStart and tod < nyStart + 30
lonTrade   = useLon and tod >= lonStart + 30 and h < lonEnd
nyTrade    = useNy and tod >= nyStart + 30 and h < nyEnd

var float rHi = na
var float rLo = na
var int   sess = 0
var bool  traded = false
var float ent = na
var float sl = na
var float tp = na
var bool  beDone = false
var int   nEntries = 0

// Construction du range
if inLonRange or inNyRange
    newSess = inLonRange ? 1 : 2
    if sess != newSess or not (inLonRange[1] or inNyRange[1])
        rHi := high
        rLo := low
        sess := newSess
        traded := false
    else
        rHi := math.max(rHi, high)
        rLo := math.min(rLo, low)

flat = strategy.position_size == 0
canTrade = flat and not traded and not na(rHi) and ((sess == 1 and lonTrade) or (sess == 2 and nyTrade))

if canTrade
    mid = (rHi + rLo) / 2
    if close > rHi and (not useTrend or dTrend == 1)
        stopLvl = stopMid ? mid : rLo
        stopLvl := math.min(stopLvl, close - minStopA * atr)
        risk = close - stopLvl
        ent := close
        sl := stopLvl
        tp := close + tpR * risk
        beDone := false
        traded := true
        nEntries += 1
        strategy.entry("L", strategy.long, qty = riskCash / (risk * syminfo.pointvalue))
        strategy.exit("LX", "L", stop = sl, limit = tp)
    else if close < rLo and (not useTrend or dTrend == -1)
        stopLvl = stopMid ? mid : rHi
        stopLvl := math.max(stopLvl, close + minStopA * atr)
        risk = stopLvl - close
        ent := close
        sl := stopLvl
        tp := close - tpR * risk
        beDone := false
        traded := true
        nEntries += 1
        strategy.entry("S", strategy.short, qty = riskCash / (risk * syminfo.pointvalue))
        strategy.exit("SX", "S", stop = sl, limit = tp)

// Stop au point d'entrée
if strategy.position_size > 0 and not beDone and high >= ent + beR * (ent - sl)
    beDone := true
    strategy.exit("LX", "L", stop = ent, limit = tp)
if strategy.position_size < 0 and not beDone and low <= ent - beR * (sl - ent)
    beDone := true
    strategy.exit("SX", "S", stop = ent, limit = tp)

// Clôture forcée en fin de séance
if strategy.position_size != 0 and ((sess == 1 and h >= lonFlat) or (sess == 2 and h >= nyFlat))
    strategy.close_all(comment = "Fin de séance")

plot(sess > 0 ? rHi : na, "Haut du range", color.green, 1, plot.style_linebr)
plot(sess > 0 ? rLo : na, "Bas du range", color.red, 1, plot.style_linebr)
bgcolor(inLonRange or inNyRange ? color.new(color.blue, 90) : na, title = "Range d'ouverture")

var table stats = table.new(position.top_right, 2, 6, bgcolor = color.white, frame_color = color.gray, frame_width = 1)
if barstate.islast
    n = strategy.closedtrades
    totalR = strategy.netprofit / riskCash
    pf = strategy.grossloss > 0 ? strategy.grossprofit / strategy.grossloss : na
    table.cell(stats, 0, 0, "Entrées", text_halign = text.align_left)
    table.cell(stats, 1, 0, str.tostring(nEntries))
    table.cell(stats, 0, 1, "Résultat total (R, frais inclus)", text_halign = text.align_left)
    table.cell(stats, 1, 1, str.tostring(totalR, "#.##"))
    table.cell(stats, 0, 2, "Espérance par entrée (R)", text_halign = text.align_left)
    table.cell(stats, 1, 2, nEntries > 0 ? str.tostring(totalR / nEntries, "#.###") : "—")
    table.cell(stats, 0, 3, "Trades gagnants", text_halign = text.align_left)
    table.cell(stats, 1, 3, n > 0 ? str.tostring(100.0 * strategy.wintrades / n, "#.#") + " %" : "—")
    table.cell(stats, 0, 4, "Facteur de profit", text_halign = text.align_left)
    table.cell(stats, 1, 4, na(pf) ? "—" : str.tostring(pf, "#.##"))
    table.cell(stats, 0, 5, "Pire baisse (R)", text_halign = text.align_left)
    table.cell(stats, 1, 5, str.tostring(strategy.max_drawdown / riskCash, "#.##"))
