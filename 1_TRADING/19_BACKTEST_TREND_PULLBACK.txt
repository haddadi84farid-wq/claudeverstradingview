// XAU Trend Pullback — règles verrouillées (continuation de tendance, entrée sur retracement + BOS).
// 1. FLUX : Daily = état 20/10 (clôtures) ; H4 = structure (sommets / creux) qui ne doit PAS aller contre. Sinon NEUTRE.
// 2. CIBLE : 1re FVG H4 non comblée dans le sens du flux, sinon plus haut / bas de la veille, sinon dernier sommet / creux H4.
//    Elle sert de filtre : TP = 2 R, ou le 1er obstacle contraire s'il est plus proche ; pas de trade si l'espace < 1,5 R.
// 3. ENTRÉE (unité du graphique, 5 min puis 1 min) — VENTE : impulsion baissière (sommet H0 → creux L1, ≥ 1 ATR 15 min),
//    retracement 38-79 % qui forme un LH confirmé, puis CLÔTURE sous le micro-creux formé après le LH → entrée à
//    l'ouverture suivante. Annulé si le prix dépasse le LH ou après N bougies. ACHAT : exactement l'inverse.
// 4. STOP : au-delà du LH (HL) + max(0,2 ATR, spread) ; au moins 0,5 ATR 15 min (élargi), au plus 0,25 ATR JOURNALIER (sinon pas de trade :
//    2 R doivent tenir dans la moitié du mouvement moyen d'une journée, quelle que soit l'unité d'entrée).
// Aucune gestion (pas de break-even, pas de partiel). Entrées 07:00-16:00 UTC, clôture forcée à 20:00 UTC.
// Anti-regard vers le futur : données HTF = bougies CLÔTURÉES (lookahead_on + [1]), pivots utilisés une fois confirmés.
// Frais : glissement 15 ticks par ordre, commission 0,035 $ par once et par ordre, marge 1 %. Résultat en R (risque fixe).
// Les annonces ne sont pas filtrées (impossible dans le testeur) : on regardera leur effet dans la liste des trades.

// ---------- Daily : état 20/10 + veille ----------
dayData() =>
    var int st = 0
    hiL = ta.highest(high, dLong)[1]
    loL = ta.lowest(low, dLong)[1]
    hiS = ta.highest(high, dShort)[1]
    loS = ta.lowest(low, dShort)[1]
    if close > hiL
        st := 1
    else if close < loL
        st := -1
    else if st == 1 and close < loS
        st := 0
    else if st == -1 and close > hiS
        st := 0
    [st[1], high[1], low[1], ta.atr(14)[1]]
[dSt, pdh, pdl, atrD] = request.security(syminfo.tickerid, "D", dayData(), lookahead = barmerge.lookahead_on)

// ---------- H4 : structure + FVG non comblées (la plus proche de chaque côté) ----------
h4Data() =>
    ph = ta.pivothigh(high, htfPiv, htfPiv)
    pl = ta.pivotlow(low, htfPiv, htfPiv)
    hA = ta.valuewhen(not na(ph), ph, 0)
    hB = ta.valuewhen(not na(ph), ph, 1)
    lA = ta.valuewhen(not na(pl), pl, 0)
    lB = ta.valuewhen(not na(pl), pl, 1)
    st = hA > hB and lA > lB ? 1 : hA < hB and lA < lB ? -1 : 0
    var array<float> uTop = array.new<float>()
    var array<float> uBot = array.new<float>()
    var array<float> dTop = array.new<float>()
    var array<float> dBot = array.new<float>()
    var array<int>   uT   = array.new<int>()
    var array<int>   dT   = array.new<int>()
    if array.size(uTop) > 0
        for i = array.size(uTop) - 1 to 0
            if low <= array.get(uBot, i)
                array.remove(uTop, i)
                array.remove(uBot, i)
                array.remove(uT, i)
    if array.size(dTop) > 0
        for i = array.size(dTop) - 1 to 0
            if high >= array.get(dTop, i)
                array.remove(dTop, i)
                array.remove(dBot, i)
                array.remove(dT, i)
    if low > high[2]
        array.push(uTop, low)
        array.push(uBot, high[2])
        array.push(uT, time[2])
    if high < low[2]
        array.push(dTop, low[2])
        array.push(dBot, high)
        array.push(dT, time[2])
    if array.size(uTop) > 60
        array.shift(uTop)
        array.shift(uBot)
        array.shift(uT)
    if array.size(dTop) > 60
        array.shift(dTop)
        array.shift(dBot)
        array.shift(dT)
    // FVG haussière non comblée sous le prix (aimant des ventes) : son haut ; FVG baissière au-dessus : son bas
    float below  = na
    float belowB = na
    int   belowT = na
    float above  = na
    float aboveT = na
    int   aboveTm = na
    if array.size(uTop) > 0
        for i = 0 to array.size(uTop) - 1
            t = array.get(uTop, i)
            if t < close and (na(below) or t > below)
                below  := t
                belowB := array.get(uBot, i)
                belowT := array.get(uT, i)
    if array.size(dBot) > 0
        for i = 0 to array.size(dBot) - 1
            b = array.get(dBot, i)
            if b > close and (na(above) or b < above)
                above   := b
                aboveT  := array.get(dTop, i)
                aboveTm := array.get(dT, i)
    [st[1], lA[1], hA[1], below[1], above[1], belowB[1], belowT[1], aboveT[1], aboveTm[1]]
[h4St, h4Low, h4High, fvgBelow, fvgAbove, fvgBelowB, fvgBelowT, fvgAboveT, fvgAboveTm] = request.security(syminfo.tickerid, "240", h4Data(), lookahead = barmerge.lookahead_on)

// Tendance du jour = structure H1 (sommets et creux de plus en plus hauts = haussière, de plus en plus bas = baissière)
h1Data() =>
    ph = ta.pivothigh(high, 3, 3)
    pl = ta.pivotlow(low, 3, 3)
    hA = ta.valuewhen(not na(ph), ph, 0)
    hB = ta.valuewhen(not na(ph), ph, 1)
    lA = ta.valuewhen(not na(pl), pl, 0)
    lB = ta.valuewhen(not na(pl), pl, 1)
    st = hA > hB and lA > lB ? 1 : hA < hB and lA < lB ? -1 : 0
    // Sommet cassé à la clôture = haussier même si le dernier creux n'est pas encore confirmé (et inversement)
    st := close > hA and lA >= lB ? 1 : close < lA and hA <= hB ? -1 : st
    st[1]
h1St = request.security(syminfo.tickerid, "60", h1Data(), lookahead = barmerge.lookahead_on)
// Sens des plans : tendance H1 si elle est nette (réglable), sinon flux journalier

// Obstacle contraire le plus proche et cible affichée
nearestBelow(float p) =>
    float o = na
    if not na(fvgBelow) and fvgBelow < p
        o := fvgBelow
    if not na(pdl) and pdl < p and (na(o) or pdl > o)
        o := pdl
    if useSwObs and not na(h4Low) and h4Low < p and (na(o) or h4Low > o)
        o := h4Low
    o
nearestAbove(float p) =>
    float o = na
    if not na(fvgAbove) and fvgAbove > p
        o := fvgAbove
    if not na(pdh) and pdh > p and (na(o) or pdh < o)
        o := pdh
    if useSwObs and not na(h4High) and h4High > p and (na(o) or h4High < o)
        o := h4High
    o
tgtSell = not na(fvgBelow) and fvgBelow < close ? "FVG H4 " + str.tostring(fvgBelow, format.mintick) :
     not na(pdl) and pdl < close ? "PDL " + str.tostring(pdl, format.mintick) :
     not na(h4Low) and h4Low < close ? "creux H4 " + str.tostring(h4Low, format.mintick) : "aucune"
tgtBuy = not na(fvgAbove) and fvgAbove > close ? "FVG H4 " + str.tostring(fvgAbove, format.mintick) :
     not na(pdh) and pdh > close ? "PDH " + str.tostring(pdh, format.mintick) :
     not na(h4High) and h4High > close ? "sommet H4 " + str.tostring(h4High, format.mintick) : "aucune"

// ---------- Structure de l'unité d'entrée ----------
ph = ta.pivothigh(high, pivLen, pivLen)
pl = ta.pivotlow(low, pivLen, pivLen)
lowN  = ta.lowest(low, pivLen)
highN = ta.highest(high, pivLen)

var float lastPhV = na
var int   lastPhI = na
var float lastPlV = na
var int   lastPlI = na
var int   nArm = 0
var int   nBos = 0
var int   kSess = 0
var int   kPos = 0
var int   kWide = 0
var int   kRoom = 0
var int   sState = 0
var float sLH = na
var float sTrig = na
var int   sArm = na
var int   bState = 0
var float bHL = na
var float bTrig = na
var int   bArm = na

if not na(ph)
    phI = bar_index - pivLen
    if flux == -1 and not na(lastPhV) and not na(lastPlV) and lastPhI < lastPlI and lastPlI < phI
        imp = lastPhV - lastPlV
        ret = imp > 0 ? (ph - lastPlV) / imp : 0.0
        if imp >= minImpAtr * atr15 and ret >= retMin and ret <= retMax and lowN > lastPlV
            sState := 1
            nArm += 1
            sLH := ph
            sTrig := lowN
            sArm := bar_index
    lastPhV := ph
    lastPhI := phI
if not na(pl)
    plI = bar_index - pivLen
    if flux == 1 and not na(lastPhV) and not na(lastPlV) and lastPlI < lastPhI and lastPhI < plI
        imp = lastPhV - lastPlV
        ret = imp > 0 ? (lastPhV - pl) / imp : 0.0
        if imp >= minImpAtr * atr15 and ret >= retMin and ret <= retMax and highN < lastPhV
            bState := 1
            nArm += 1
            bHL := pl
            bTrig := highN
            bArm := bar_index
    lastPlV := pl
    lastPlI := plI

// ---------- Déclenchement (BOS à la clôture) ----------
var int   nEntries = 0
var float curStop = na
var float curTp = na
var string lastSkip = ""
buf = math.max(slBufAtr * atr, spreadPts)

if sState == 1 and bar_index > sArm
    if high > sLH or flux != -1 or bar_index - sArm > maxWait
        sState := 0
    else if close < sTrig
        sState := 0
        nBos += 1
        stopP = math.max(sLH + buf, close + minStopA * atr15)
        risk = stopP - close
        obst = nearestBelow(close)
        tpDist = na(obst) ? tpR * risk : math.min(tpR * risk, close - obst)
        if not inSession
            lastSkip := "hors horaires"
            kSess += 1
        else if strategy.position_size != 0
            lastSkip := "position déjà ouverte"
            kPos += 1
        else if risk > maxStopD * atrD
            lastSkip := "stop trop large"
            kWide += 1
        else if tpDist < minTpR * risk
            lastSkip := "obstacle trop proche"
            kRoom += 1
        else
            strategy.entry("SELL", strategy.short, qty = riskCash / (risk * syminfo.pointvalue))
            strategy.exit("SELL X", "SELL", stop = stopP, limit = close - tpDist)
            curStop := stopP
            curTp := close - tpDist
            nEntries += 1
if bState == 1 and bar_index > bArm
    if low < bHL or flux != 1 or bar_index - bArm > maxWait
        bState := 0
    else if close > bTrig
        bState := 0
        nBos += 1
        stopP = math.min(bHL - buf, close - minStopA * atr15)
        risk = close - stopP
        obst = nearestAbove(close)
        tpDist = na(obst) ? tpR * risk : math.min(tpR * risk, obst - close)
        if not inSession
            lastSkip := "hors horaires"
            kSess += 1
        else if strategy.position_size != 0
            lastSkip := "position déjà ouverte"
            kPos += 1
        else if risk > maxStopD * atrD
            lastSkip := "stop trop large"
            kWide += 1
        else if tpDist < minTpR * risk
            lastSkip := "obstacle trop proche"
            kRoom += 1
        else
            strategy.entry("BUY", strategy.long, qty = riskCash / (risk * syminfo.pointvalue))
            strategy.exit("BUY X", "BUY", stop = stopP, limit = close + tpDist)
            curStop := stopP
            curTp := close + tpDist
            nEntries += 1

if strategy.position_size != 0 and hUtc >= closeH
    strategy.close_all(comment = "Fin de journée")

inPos = strategy.position_size != 0
plot(inPos ? curStop : na, "Stop", cRed, 2, plot.style_linebr)

plot(sState == 1 ? sTrig : bState == 1 ? bTrig : na, "Niveau de cassure (BOS)", cGray, 1, plot.style_linebr)

// ---------- Scénario dessiné : 1. rebond → 2. zone de retracement 38-62 % (pâle) et OTE 62-79 % (vive) de la dernière impulsion + Fibonacci décalé à droite → 3. cible ----------
// Dernière impulsion calculée en 15 MIN, quelle que soit l'unité affichée (même zone et même Fibonacci en 1, 5 ou 15 min)
impData() =>
    hb = ta.highestbars(high, impLen)
    lb = ta.lowestbars(low, impLen)
    [ta.highest(high, impLen), ta.lowest(low, math.max(1, 1 - nz(hb))), ta.lowest(low, impLen), ta.highest(high, math.max(1, 1 - nz(lb)))]
[impHi, lowSinceHi, impLo, highSinceLo] = request.security(syminfo.tickerid, "15", impData())
var array<line>  fibL = array.new<line>()
var array<label> fibT = array.new<label>()
var array<box>   plB  = array.new<box>()
var array<label> plT  = array.new<label>()
var array<line>  plL  = array.new<line>()
H = 3600000
fibLvl(float e, float sg, float rg, float r) => e + sg * r * rg
// Taille en lots pour un risque donné en € (1 lot = 100 onces), minimum 0,01
lotsFor(float dist) =>
    riskPerLot = dist * 100 / eurUsd
    math.max(0.01, math.floor(riskEur / riskPerLot * 100) / 100)
// Un niveau du plan : trait + étiquette posée juste au-dessus du trait, À L'INTÉRIEUR du dessin (côté gauche)
planLine(int t0, int t1, float p, string txt, color c, int w) =>
    array.push(plL, line.new(t0, p, t1, p, xloc = xloc.bar_time, color = c, width = w))
    array.push(plT, label.new(t0 + math.round((t1 - t0) * 0.03), p, txt, xloc = xloc.bar_time, style = label.style_label_lower_left,
         color = color.new(c, 100), textcolor = c == color.white ? color.white : c, size = size.small))
// Un plan complet, dessiné de la même façon pour A et B :
// flèche prix → entrée, boîte risque (rouge) et gain (vert), traits SL / Entrée / TP1-TP2-TP3 (1, 2, 3 R, avant la cible) / Cible, flèche entrée → cible
drawPlan(int t0, int t1, float e, float sl, float tgt, string name, color c) =>
    dist = math.abs(e - sl)
    up   = tgt > e
    lots = lotsFor(dist)
    riskReal = lots * dist * 100 / eurUsd
    array.push(plL, line.new(time, close, t0, e, xloc = xloc.bar_time, color = color.new(c, 40), width = 2, style = line.style_arrow_right))
    array.push(plB, box.new(t0, math.max(e, sl), t1, math.min(e, sl), xloc = xloc.bar_time, border_color = color.new(cRed, 60), bgcolor = color.new(cRed, 82)))
    array.push(plB, box.new(t0, math.max(e, tgt), t1, math.min(e, tgt), xloc = xloc.bar_time, border_color = color.new(cGreen, 60), bgcolor = color.new(cGreen, 86)))
    array.push(plT, label.new(t0, math.min(sl, tgt, e), name + " | " + str.tostring(lots, "#.##") + " lot | risque " + str.tostring(riskReal, "#") + " €",
         xloc = xloc.bar_time, style = label.style_label_upper_left, color = c, textcolor = color.white, size = size.small))
    planLine(t0, t1, sl, "SL " + str.tostring(sl, format.mintick), cRed, 2)
    planLine(t0, t1, e, "Entrée " + str.tostring(e, format.mintick), color.white, 2)
    n = 0
    for k = 1 to 3
        tp = e + (up ? 1 : -1) * k * dist
        if up ? tp < tgt : tp > tgt
            n += 1
            planLine(t0, t1, tp, "TP" + str.tostring(k) + " " + str.tostring(tp, format.mintick) + " (" + str.tostring(k) + " R)", cGreen, 1)
    planLine(t0, t1, tgt, "TP" + str.tostring(n + 1) + " = CIBLE " + str.tostring(tgt, format.mintick) + " (" + str.tostring(math.abs(tgt - e) / dist, "#.#") + " R)", cGreen, 2)
    array.push(plL, line.new(t1, e, t1 + H, tgt, xloc = xloc.bar_time, color = c, width = 2, style = line.style_arrow_right))
if barstate.islast

        line.delete(l)

        label.delete(l)
    for b in plB

        float tgt = na
        if isS
            tgt := not na(fvgBelow) and fvgBelow < close ? fvgBelow : not na(pdl) and pdl < close ? pdl : enV
        else
            tgt := not na(fvgAbove) and fvgAbove > close ? fvgAbove : not na(pdh) and pdh > close ? pdh : enV

        // FVG H4 visée (la cible) : rectangle depuis sa création jusqu'à la droite des plans

                 border_color = color.new(color.orange, 20), bgcolor = color.new(color.orange, 80),
                 text = "FVG H4 (cible) " + str.tostring(math.min(fTop, fBot), "#") + "–" + str.tostring(math.max(fTop, fBot), "#"),
                 text_color = color.orange, text_size = size.small, text_halign = text.align_left))
        // PLAN B (rebond vers l'OTE, contre le flux) : limite au 38,2 %, stop au-delà + marge, cible = début de l'OTE (61,8 %)
        eB   = fibLvl(enV, sg, rng, 0.382)
        slB  = eB - sg * pbuf
        tgtB = fibLvl(enV, sg, rng, 0.618)
        okB  = isS ? close > eB and close < tgtB : close < eB and close > tgtB
        okB := okB and math.abs(tgtB - eB) / math.abs(eB - slB) >= 1.5
        // PLAN A (dans le sens du flux) : limite au milieu de l'OTE (70,5 %), stop au-delà de 78,6 % + marge, cible = FVG H4 / veille
        eA  = fibLvl(enV, sg, rng, 0.705)
        slA = fibLvl(enV, sg, rng, 0.786) + sg * pbuf
        tStart = time + math.round(planStart * H)
        drawB = showB and okB
        rrB = math.abs(tgtB - eB) / math.abs(eB - slB)
        inZoneB = isS ? close > eB and close < tgtB : close < eB and close > tgtB
        if drawB
            drawPlan(tStart, tStart + 3 * H, eB, slB, tgtB, isS ? "PLAN B — ACHAT limite (rebond vers l'OTE, contre le sens)" : "PLAN B — VENTE limite (repli vers l'OTE, contre le sens)", color.new(color.orange, 0))
        else if showB
            // Plan B coché mais pas valable : on le dit au lieu de ne rien afficher
            whyB = not inZoneB ? "le prix n'est pas entre 38,2 % (" + str.tostring(eB, "#") + ") et 61,8 % (" + str.tostring(tgtB, "#") + ")" : "R:R " + str.tostring(rrB, "#.#") + " < 1,5"
            array.push(plT, label.new(tStart, close, "PLAN B : pas de trade maintenant\n" + whyB, xloc = xloc.bar_time, style = label.style_label_left,
                 color = color.new(cGray, 30), textcolor = color.white, size = size.small))
        if showA
            tA0 = showB ? tStart + 5 * H : tStart
            lvlX = fibLvl(enV, sg, rng, 0.786)
            // Annulé si une bougie 15 min a CLÔTURÉ au-delà de 78,6 % (le rebond n'est plus un simple retracement)
            killA = isS ? c15 > lvlX : c15 < lvlX
            // Confirmation : dernière bougie 15 min qui a touché l'OTE et clôturé dans le sens du flux (rejet)
            confA = isS ? h15 >= fibLvl(enV, sg, rng, 0.618) and c15 < o15 and c15 < eA : l15 <= fibLvl(enV, sg, rng, 0.618) and c15 > o15 and c15 > eA
            modeTxt = modeA == "Limite" ? (isS ? "VENTE limite" : "ACHAT limite") : (isS ? "VENTE sur confirmation" : "ACHAT sur confirmation")
            stTxt = killA ? "ANNULÉ (clôture 15 min au-delà de 78,6 %)" : modeA == "Limite" ? "ordre limite actif" : confA ? "CONFIRMÉ : tu peux entrer" : "EN ATTENTE du rejet"
            condTxt = modeA == "Limite" ? "" : (isS ? "\nEntrer seulement si une bougie 15 min touche l'OTE puis clôture ROUGE sous l'entrée" : "\nEntrer seulement si une bougie 15 min touche l'OTE puis clôture VERTE au-dessus de l'entrée")
            drawPlan(tA0, tA0 + 3 * H, eA, slA, tgt, "PLAN A — " + modeTxt + " (OTE) — " + stTxt + (contre ? "\n⚠ Dans le sens du H1 mais CONTRE le flux journalier : taille ÷ 2" : "") + condTxt, killA ? cGray : cZ)
            planLine(tA0, tA0 + 3 * H, lvlX, "ANNULATION si clôture 15 min au-delà de " + str.tostring(lvlX, format.mintick), color.orange, 1)
        // Fibonacci de la dernière impulsion, dessiné à l'écart (décalé vers la droite)
        if showFib
            f0 = time + math.round(fibShift * H)
            f1 = f0 + 4 * H
            lv = array.from(0.0, 0.382, 0.5, 0.618, 0.786, 1.0)
            for k = 0 to array.size(lv) - 1
                r = array.get(lv, k)
                p = fibLvl(enV, sg, rng, r)
                isOte = r == 0.618 or r == 0.786
                c = r == 0.0 or r == 1.0 ? color.white : isOte ? cZ : cGray
                array.push(fibL, line.new(f0, p, f1, p, xloc = xloc.bar_time, color = c, width = isOte ? 2 : 1,
                     style = isOte ? line.style_solid : line.style_dashed))
                array.push(fibT, label.new(f1, p, str.tostring(r * 100, "#.#") + " %  " + str.tostring(p, format.mintick),
                     xloc = xloc.bar_time, style = label.style_label_left, color = color.new(c, 100), textcolor = c, size = size.small))
