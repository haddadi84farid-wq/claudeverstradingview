//@version=6
// Backtest : CASSURE du plus haut / plus bas de la veille (PDH / PDL) en CONTINUATION.
// Idée : le backtest « balayage » montre que CONTRER un PDH/PDL perd (−0,40 R par trade). On teste l'inverse.
// Règles :
// - Cassure acceptée : 2 clôtures consécutives au-delà du PDH (achat) ou du PDL (vente), entre 07:00 et 18:00 UTC.
// - Entrée : ordre limite au RETEST du niveau cassé (PDH pour un achat, PDL pour une vente), valable N bougies.
//   Option « entrée directe » : au marché à la 2e clôture, sans attendre le retest.
// - Stop : de l'autre côté du niveau, à « marge ATR » × ATR 15 min + spread. TP : « TP en R ». Stop au point d'entrée à 1 R.
// - Une seule entrée par sens et par jour, 07:00-18:00 UTC, clôture forcée à 20:00 UTC.
// Chaque trade risque le même montant : Résultat ÷ risque = résultat en R (tableau en haut à droite).
strategy("Backtest XAU — cassure PDH/PDL", overlay = true, initial_capital = 10000,
     default_qty_type = strategy.fixed, default_qty_value = 1, slippage = 10, pyramiding = 0,
     margin_long = 1, margin_short = 1)

riskCash   = input.float(100.0, "Risque par trade ($)", minval = 1)
spreadPts  = input.float(0.3, "Spread (points)", minval = 0)
atrMult    = input.float(1.0, "Marge du stop (× ATR)", minval = 0.1)
tpR        = input.float(2.0, "TP en R", minval = 0.5)
beR        = input.float(1.0, "Stop au point d'entrée à (R)", minval = 0.3)
retestBars = input.int(12, "Bougies max pour le retest", minval = 1, maxval = 48)
direct     = input.bool(false, "Entrée directe (sans retest)")
startH     = input.int(7, "Début des entrées (heure UTC)", minval = 0, maxval = 23)
endH       = input.int(18, "Fin des entrées (heure UTC)", minval = 1, maxval = 24)
flatH      = input.int(20, "Clôture forcée (heure UTC)", minval = 1, maxval = 23)

atr    = ta.atr(14)
h      = hour(time, "UTC")
newDay = ta.change(dayofmonth(time, "UTC")) != 0
[pdh, pdl] = request.security(syminfo.tickerid, "D", [high[1], low[1]], lookahead = barmerge.lookahead_on)
inWindow = h >= startH and h < endH

var bool  doneL = false
var bool  doneS = false
var float armL  = na
var int   armLBar = 0
var float armS  = na
var int   armSBar = 0
var float sl = na
var float tp = na
var float ent = na
var bool  beDone = false
var int   nEntries = 0

if newDay
    doneL := false
    doneS := false
    armL := na
    armS := na

// Cassure acceptée : 2 clôtures consécutives au-delà du niveau (la précédente et celle-ci)
brkL = close > pdh and close[1] > pdh and close[2] <= pdh
brkS = close < pdl and close[1] < pdl and close[2] >= pdl

flat = strategy.position_size == 0

if inWindow and flat and not doneL and brkL
    if direct
        risk = close - (pdh - atrMult * atr - spreadPts)
        if risk > 0
            ent := close
            sl := pdh - atrMult * atr - spreadPts
            tp := ent + tpR * risk
            beDone := false
            strategy.entry("L", strategy.long, qty = riskCash / (risk * syminfo.pointvalue))
            strategy.exit("LX", "L", stop = sl, limit = tp)
            doneL := true
            nEntries += 1
    else
        armL := pdh
        armLBar := bar_index

if inWindow and flat and not doneS and brkS
    if direct
        risk = (pdl + atrMult * atr + spreadPts) - close
        if risk > 0
            ent := close
            sl := pdl + atrMult * atr + spreadPts
            tp := ent - tpR * risk
            beDone := false
            strategy.entry("S", strategy.short, qty = riskCash / (risk * syminfo.pointvalue))
            strategy.exit("SX", "S", stop = sl, limit = tp)
            doneS := true
            nEntries += 1
    else
        armS := pdl
        armSBar := bar_index

// Retest : ordre limite au niveau cassé (rempli si la bougie suivante le touche)
if not na(armL)
    if bar_index - armLBar > retestBars or not inWindow or doneL or close < armL - atrMult * atr
        armL := na
        strategy.cancel("L")
    else if flat and bar_index > armLBar
        ent := armL
        sl := armL - atrMult * atr - spreadPts
        tp := ent + tpR * (ent - sl)
        beDone := false
        strategy.entry("L", strategy.long, qty = riskCash / ((ent - sl) * syminfo.pointvalue), limit = ent)
        strategy.exit("LX", "L", stop = sl, limit = tp)

if not na(armS)
    if bar_index - armSBar > retestBars or not inWindow or doneS or close > armS + atrMult * atr
        armS := na
        strategy.cancel("S")
    else if flat and bar_index > armSBar
        ent := armS
        sl := armS + atrMult * atr + spreadPts
        tp := ent - tpR * (sl - ent)
        beDone := false
        strategy.entry("S", strategy.short, qty = riskCash / ((sl - ent) * syminfo.pointvalue), limit = ent)
        strategy.exit("SX", "S", stop = sl, limit = tp)

// Entrée remplie : on compte, on arrête d'armer
if strategy.position_size > 0 and strategy.position_size[1] == 0
    doneL := true
    armL := na
    nEntries += direct ? 0 : 1
if strategy.position_size < 0 and strategy.position_size[1] == 0
    doneS := true
    armS := na
    nEntries += direct ? 0 : 1

// Stop au point d'entrée à beR
if strategy.position_size > 0 and not beDone and high >= ent + beR * (ent - sl)
    beDone := true
    strategy.exit("LX", "L", stop = ent, limit = tp)
if strategy.position_size < 0 and not beDone and low <= ent - beR * (sl - ent)
    beDone := true
    strategy.exit("SX", "S", stop = ent, limit = tp)

if h >= flatH and strategy.position_size != 0
    strategy.close_all(comment = "Fin de journée")

plot(pdh, "PDH", color.red, 1, plot.style_linebr)
plot(pdl, "PDL", color.green, 1, plot.style_linebr)

var table stats = table.new(position.top_right, 2, 4, bgcolor = color.white, frame_color = color.gray, frame_width = 1)
if barstate.islast
    totalR = strategy.netprofit / riskCash
    table.cell(stats, 0, 0, "Entrées", text_halign = text.align_left)
    table.cell(stats, 1, 0, str.tostring(nEntries))
    table.cell(stats, 0, 1, "Résultat total (R)", text_halign = text.align_left)
    table.cell(stats, 1, 1, str.tostring(totalR, "#.##"))
    table.cell(stats, 0, 2, "Espérance par entrée (R)", text_halign = text.align_left)
    table.cell(stats, 1, 2, nEntries > 0 ? str.tostring(totalR / nEntries, "#.###") : "—")
    table.cell(stats, 0, 3, "Pire baisse (R)", text_halign = text.align_left)
    table.cell(stats, 1, 3, str.tostring(strategy.max_drawdown / riskCash, "#.##"))
